:Driving_Loop
while $OnCarMission == 0
    wait 250
    if or
        not Player.IsPlaying($player)
        Camera.GetFadingStatus()
    then
        continue
    end

    if and
        $onmission == false
        $Intro_Passed == false
    then
        if and
            Player.CanStartMission($player)
            Player.LocateOnFoot3D($player, 774.45, -880.88, 14.74, 3.0, 3.0, 3.0, False)
        then
            Blip.Remove($MissionTriggerBlip)
            Sphere.Remove($MissionTriggerSphere)
            Player.SetControl($player, False)
            Text.PrintBigString("Nowhere Like Home...", 5000, TextStyle.BottomRight)
            Camera.SetFadingColor(40, 0, 70)
            Camera.DoFade(1500, Fade.Out)
            wait 1500
            $OnCarMission = 1
            stream_custom_script "INIT.cm"
            break
        end
        if and
            0@ == 0
            Player.LocateAnyMeans3D($player, 774.45, -880.88, 14.74, 17.0, 17.0, 10.0, False)
        then
            Text.Print('MARKERW', 3000, 0) // Walk into the blue marker to continue.
            0@ = 1
        end
    else
        if and
            $onmission == 1
            Player.LocateOnFoot3D($player, 774.45, -880.88, 14.74, 3.0, 3.0, 3.0, False)
        then
            Text.Print('FUCKOFF', 4000, 1)
            wait 5000
        end
    end
end

while $OnCarMission == 1
    wait 250
end

if
    $CarMissionDone == 1

    start_new_script @Pistol

    $MissionTriggerSphere = Sphere.Create(774.45, -880.88, 14.74, 3.0)
    goto @Driving_Loop
end

/// Trigger mission 2: "Finding Your Bearings..."
:Pistol
script_name 'Pistol'
wait 500
Text.Print('JUNKYRD', 5000, 0)
$MissionTriggerBlip = Blip.AddSpriteForContactPoint(1120.63, 27.91, 2.12, RadarSprite.Joey)
$MissionTriggerSphere = Sphere.Create(1120.63, 27.91, 2.12, 3.0)

:Pistol_Loop
while $OnGunMission == 0
    wait 250
    if or
        not Player.IsPlaying($player)
        Camera.GetFadingStatus()
    then
        continue
    end

    // Heat Check Logic
    if and
        Player.LocateAnyMeans3D($player, 1120.63, 27.91, 2.12, 3.0, 3.0, 3.0, False)
        Player.IsWantedLevelGreater($player, 0)
    then
        if
            $SprayShop_Active == 0
        then
            Text.Print('HEAT', 4000, 1)
            $SprayShopIND = Blip.AddSpriteForCoord(924.0, -361.0, 10.0, RadarSprite.Spray)
            $SprayShop_Active = 1
        end
    else
        if
            $SprayShop_Active == 1
        then
            Blip.Remove($SprayShopIND)
            $SprayShop_Active = 0
        end
    end

    // Trigger Logic
    if and
        Player.CanStartMission($player)
        $onmission == 0
        not Player.IsWantedLevelGreater($player, 0)
        Player.LocateOnFoot3D($player, 1120.63, 27.91, 2.12, 3.0, 3.0, 2.0, False)
    then
        Blip.Remove($MissionTriggerBlip)
        Sphere.Remove($MissionTriggerSphere)
        if $SprayShop_Active == 1
        then
            Blip.Remove($SprayShopIND)
            $SprayShop_Active = 0
        end
        Player.SetControl($player, False)
        Text.PrintBigString("Finding Your Bearings...", 5000, TextStyle.BottomRight)
        Camera.SetFadingColor(40, 0, 70)
        Camera.DoFade(1500, Fade.Out)
        wait 1500
        $OnGunMission = 1
        stream_custom_script "pistol.cm"
        break
    end
end

// Wait for the mission to complete or fail
while $OnGunMission == 1
    wait 250
end

// Post-Mission Logic
if
    $SafeAndSound == 1
then
    start_new_script @Stairs
    terminate_this_script
else
    $OnGunMission = 0
    goto @Pistol
end

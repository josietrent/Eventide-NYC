:CarMarie

while true
    wait 0

    if
        not Char.IsStillAlive(Marie)
    then
        jump @Marie_Fail
    end

    if
        Player.IsInAnyCar($player)
    then
        Player.StoreCarIsInNoSave($player, MyCar)
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
    else
        if and
            CarBlip == 0
            Car.IsStillAlive(MyCar)
        then
            CarBlip = Blip.AddForCar(MyCar)
        end
    end

    if
        not Car.IsStillAlive(MyCar)
    then
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
        if or
            NeedsCar == 0
            TIMERA >= 3000
        then
            Text.Print('WHEELS', 3000, 0)
            TIMERA = 0
            NeedsCar = 1
        end
        continue
    end
    NeedsCar = 0

    if
        Char.IsInCar(Marie, MyCar)
    then
        break
    end

    // 1. Health Check - Prevent her from entering a wrecked or burning vehicle
    Car.GetHealth(MyCar, $CarHealth)
    if
        $CarHealth < 600
    then
        if
            $CarHealth < 250
        then
            Char.SetObjFleeCar(Marie, MyCar)
        else
            Char.SetObjWaitOnFoot(Marie)
        end
        if
            $SprayShop_Active == 0
        then
            Text.Print('MISWREK', 3000, 0)
            $SprayShopIND = Blip.AddSpriteForCoord(924.0, -361.0, 10.0, RadarSprite.Spray)
            $SprayShop_Active = 1
        end
    else
        if
            $SprayShop_Active == 1
        then
            Blip.Remove($SprayShopIND)
            $SprayShop_Active = 0
        end

        // 2. Proximity check - Task her to enter ONLY if car is healthy
        if
            Char.LocateAnyMeansCar3D(Marie, MyCar, 20.0, 20.0, 20.0, False)
        then
            Char.SetObjEnterCarAsPassenger(Marie, MyCar)
            Char.SetRunning(Marie, True)
            break
        else
            Text.Print('MISFIND', 3000, 0)
            Char.SetObjWaitOnFoot(Marie)
            FindMarie = 1
            MarieBlip = Blip.AddForChar(Marie)
            Blip.ChangeDisplay(MarieBlip, BlipDisplay.BlipOnly)
            Blip.ChangeColor(MarieBlip, BlipColor.Cyan)
            jump @FindMarie
        end
    end
end

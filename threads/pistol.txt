:CarMisty
NeedsCar = 0
while true
    wait 0

    if
        not Char.IsStillAlive(Misty)
    then
        jump @GunFail
    end

    if
        Player.IsInAnyCar($player)
    then
        Player.StoreCarIsInNoSave($player, MyCar)
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
    else
        if and
            CarBlip == 0
            Car.IsStillAlive(MyCar)
        then
            CarBlip = Blip.AddForCar(MyCar)
        end
    end

    if
        not Car.IsStillAlive(MyCar)
    then
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
        if or
            NeedsCar == 0
            TIMERA >= 3000
        then
            Text.Print('WHEELS', 3000, 0)
            TIMERA = 0
            NeedsCar = 1
        end
        continue
    end
    NeedsCar = 0

    if
        Char.IsInCar(Misty, MyCar)
    then
        break
    end

    // 1. Health Check - Prevent her from entering a wrecked or burning vehicle
    Car.GetHealth(MyCar, $CarHealth)
    if
        $CarHealth < 600
    then
        Text.Print('MISWREK', 1000, 0)
        if
            $CarHealth < 250
        then
            Char.SetObjFleeCar(Misty, MyCar)
        else
            Char.SetObjWaitOnFoot(Misty)
        end
        if
            $SprayShop_Active == 0
        then
            $SprayShopIND = Blip.AddSpriteForCoord(924.0, -361.0, 10.0, RadarSprite.Spray)
            $SprayShop_Active = 1
        end
    else
        if
            $SprayShop_Active == 1
        then
            Blip.Remove($SprayShopIND)
            $SprayShop_Active = 0
        end

        // 2. Proximity check - Task her to enter ONLY if car is healthy
        if
            Char.LocateAnyMeansCar3D(Misty, MyCar, 20.0, 20.0, 20.0, False)
        then
            Char.SetObjEnterCarAsPassenger(Misty, MyCar)
            Char.SetRunning(Misty, True)
            break
        else
            Text.Print('MISFIND', 3000, 0)
            Char.SetObjWaitOnFoot(Misty)
            FindMisty = 1
            MistyBlip = Blip.AddForChar(Misty)
            Blip.ChangeDisplay(MistyBlip, BlipDisplay.BlipOnly)
            Blip.ChangeColor(MistyBlip, BlipColor.Cyan)
            jump @FindMisty
        end
    end
end

:Arrived
while true
    wait 0
    if
        not Char.IsStillAlive(Misty)
    then
        jump @GunFail
    end

    // 1. Car dead or on fire (< 250 HP): bail out & flee before it explodes
    if
        not Car.IsStillAlive(MyCar)
    then
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
        Blip.Remove(Destination)
        Sphere.Remove(GunSphere)
        if
            Char.IsInCar(Misty, MyCar)
        then
            Char.SetObjLeaveCar(Misty, MyCar)
        end
        Char.SetObjFleeOnFootTillSafe(Misty)
        TIMERA = 0
        while TIMERA < 1500
            wait 0
            if
                not Char.IsStillAlive(Misty)
            then
                jump @GunFail
            end
        end
        jump @CarMisty
    end

    Car.GetHealth(MyCar, $CarHealth)
    if
        $CarHealth < 250
    then
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
        Blip.Remove(Destination)
        Sphere.Remove(GunSphere)
        Text.Print('MISWREK', 3000, 0)
        if
            Char.IsInCar(Misty, MyCar)
        then
            Char.SetObjLeaveCar(Misty, MyCar)
            while Char.IsInCar(Misty, MyCar)
                wait 0
                if
                    not Char.IsStillAlive(Misty)
                then
                    jump @GunFail
                end
                if
                    not Car.IsStillAlive(MyCar)
                then
                    break
                end
            end
        end
        if
            Car.IsStillAlive(MyCar)
        then
            Char.SetObjFleeCar(Misty, MyCar)
        else
            Char.SetObjFleeOnFootTillSafe(Misty)
        end
        TIMERA = 0
        while TIMERA < 2000
            wait 0
            if
                not Char.IsStillAlive(Misty)
            then
                jump @GunFail
            end
        end
        jump @CarMisty
    end

    // 2. Car wrecked (< 600 HP): she refuses to ride in it and gets out
    if
        $CarHealth < 600
    then
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
        Blip.Remove(Destination)
        Sphere.Remove(GunSphere)
        Text.Print('MISWREK', 3000, 0)
        if
            Char.IsInCar(Misty, MyCar)
        then
            Char.SetObjLeaveCar(Misty, MyCar)
            while Char.IsInCar(Misty, MyCar)
                wait 0
                if
                    not Char.IsStillAlive(Misty)
                then
                    jump @GunFail
                end
                if
                    not Car.IsStillAlive(MyCar)
                then
                    break
                end
            end
        end
        jump @CarMisty
    end

    // 3. Player switched into a different car: Misty exits old car and follows to new one
    if and
        Player.IsInAnyCar($player)
        not Player.IsInCar($player, MyCar)
    then
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
        Blip.Remove(Destination)
        Sphere.Remove(GunSphere)
        if
            Char.IsInCar(Misty, MyCar)
        then
            Char.SetObjLeaveCar(Misty, MyCar)
            while Char.IsInCar(Misty, MyCar)
                wait 0
                if
                    not Char.IsStillAlive(Misty)
                then
                    jump @GunFail
                end
                if
                    not Car.IsStillAlive(MyCar)
                then
                    break
                end
            end
        end
        Player.StoreCarIsInNoSave($player, MyCar)
        jump @CarMisty
    end

    // 4. Misty exited/pulled out while car is still healthy
    if not Char.IsInCar(Misty, MyCar)
    then
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
        Blip.Remove(Destination)
        Sphere.Remove(GunSphere)
        MistyBlip = Blip.AddForChar(Misty)
        Blip.ChangeColor(MistyBlip, BlipColor.Cyan)
        Text.Print('MISWHRE', 3000, 0)
        Text.PrintSoon('NOCHASE', 5000, 0)

        $TIMER = 20000 // Give the player 20 seconds to return
        Hud.DisplayTimer($TIMER)

        while not Char.IsInCar(Misty, MyCar)
            wait 0

            if
                not Char.IsStillAlive(Misty)
            then
                jump @GunFail
            end

            if
                Player.IsInAnyCar($player)
            then
                Player.StoreCarIsInNoSave($player, MyCar)
            end

            if
                not Car.IsStillAlive(MyCar)
            then
                Hud.ClearTimer($TIMER)
                Blip.Remove(MistyBlip)
                jump @CarMisty
            end

            Car.GetHealth(MyCar, $CarHealth)
            if
                $CarHealth < 600
            then
                Hud.ClearTimer($TIMER)
                Blip.Remove(MistyBlip)
                jump @CarMisty
            end

            // If car is close & healthy, task her to enter
            if
                Char.LocateAnyMeansCar3D(Misty, MyCar, 10.0, 10.0, 10.0, False)
            then
                Char.SetObjEnterCarAsPassenger(Misty, MyCar)
            end

            if
                $TIMER == 0
            then
                Text.Print('MISTAXI', 3000, 1)
                jump @GunFail
            end
        end

        // She's back! Cleanup and restore destination if player is in MyCar
        Hud.ClearTimer($TIMER)
        Blip.Remove(MistyBlip)
        if
            Player.IsInCar($player, MyCar)
        then
            Destination = Blip.AddForCoord(1147.61, -95.21, 7.02)
            Blip.ChangeColor(Destination, BlipColor.Yellow)
            GunSphere = Sphere.Create(1147.61, -95.21, 7.02, 3.0)
        end
    end

    // 5. Player exited MyCar on foot: blip the car they were just in
    if
        not Player.IsInAnyCar($player)
    then
        if
            CarBlip == 0
        then
            Blip.Remove(Destination)
            Sphere.Remove(GunSphere)
            CarBlip = Blip.AddForCar(MyCar)
            Text.Print('INCAR', 3000, 0)
        end
    else
        if
            CarBlip > 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
            Destination = Blip.AddForCoord(1147.61, -95.21, 7.02)
            Blip.ChangeColor(Destination, BlipColor.Yellow)
            GunSphere = Sphere.Create(1147.61, -95.21, 7.02, 3.0)
        end
    end

    if and
        Player.IsInCar($player, MyCar)
        Char.IsInCar(Misty, MyCar)
        Player.LocateAnyMeans3D($player, 1147.61, -95.21, 7.02, 1.0, 1.0, 0.75, False)
    then
        Player.SetControl($player, False)
        Blip.Remove(Destination)
        Sphere.Remove(GunSphere)
        Text.PrintNow('MISRIDE', 3000, 0)
        wait 3000
        Text.Print('MISLUI', 4000, 0)
        wait 4000
        Text.Print('MISLUI2', 5000, 0)
        wait 5000
        Text.Print('MISLUI3', 5000, 0)
        wait 5000
        Text.Print('MISMONY', 5000, 0)
        Player.AddScore($player, 2000)
        wait 5000
        Text.Print('MISWORK', 8000, 0)
        wait 5500
        Char.SetObjLeaveCar(Misty, MyCar)
        Char.SetObjGotoCoordOnFoot(Misty, 1171.91, -95.28)
        Char.SetRunning(Misty, True)
        while Char.IsOnScreen(Misty)
            wait 0
            if or
                Char.LocateAnyMeans2D(Misty, 1171.91, -95.28, 2.0, 2.0, False)
                not Char.IsOnScreen(Misty)
            then
                Char.RemoveElegantly(Misty)
                Player.SetControl($player, True)
                Text.PrintHelp('HIDEOUT')
                Destination = Blip.AddForCoord(1419.15, -194.40, 49.74)
                Blip.ChangeColor(Destination, BlipColor.Yellow)
                GunSphere = Sphere.Create(1419.15, -194.40, 49.74, 3.0)
                jump @Gun_Success
            end
        end
    end
end

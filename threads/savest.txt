// Staunton Island Condo hideout save pickup & cross-island Savehouse blip visibility thread
{$CLEO .cs}
{$USE CLEO}
{$INCLUDE ../utils/globals.txt}
nop

script_name 'SaveST'

while not Player.IsPlaying($player)
    wait 250
end

// Wait for initial fade-in / save restore to settle
while true
    wait 100
    if and
        not Camera.GetFadingStatus()
        Game.HasSaveGameFinished()
    then
        break
    end
end

while true
    wait 0

    if or
        Camera.GetFadingStatus()
        not Player.IsPlaying($player)
    then
        continue
    end

    // Portland Savehouse blip: visible only while Industrial collision is loaded (and Commercial is not)
    if and
        $Intro_Passed == 1
        Streaming.IsCollisionInMemory(LevelName.Industrial)
        not Streaming.IsCollisionInMemory(LevelName.Commercial)
    then
        if $SafeBlip == 0
        then
            $SafeBlip = Blip.AddSpriteForContactPoint(1454.08, -173.18, 55.77, RadarSprite.Savehouse)
        end
    else
        if not $SafeBlip == 0
        then
            Blip.Remove($SafeBlip)
            $SafeBlip = 0
        end
    end

    // Staunton Savehouse blip: visible only while Staunton is unlocked and Commercial collision is loaded (and Industrial is not)
    if and
        $BarrierIND == 0
        Streaming.IsCollisionInMemory(LevelName.Commercial)
        not Streaming.IsCollisionInMemory(LevelName.Industrial)
    then
        if $SafeBlipST == 0
        then
            $SafeBlipST = Blip.AddSpriteForContactPoint(513.55, -627.24, 16.66, RadarSprite.Savehouse)
        end
    else
        if not $SafeBlipST == 0
        then
            Blip.Remove($SafeBlipST)
            $SafeBlipST = 0
        end
        continue
    end

    if $onmission == true
    then
        continue
    end

    if and
        Player.LocateAnyMeans3D($player, 513.55, -627.24, 16.66, 1.0, 2.0, 0.75, True)
        Game.HasSaveGameFinished()
    then
        Game.ActivateSaveMenu()
        Player.SetControl($player, False)
        Player.SetCoordinates($player, 516.50, -627.24, 15.62)
        Player.SetHeading($player, 270.0)

        :WAIT_FOR_SAVE_ST
        wait 0
        if
            Game.HasSaveGameFinished()
        then
            wait 250
            Player.SetControl($player, True)
        else
            jump @WAIT_FOR_SAVE_ST
        end
        continue
    end
end

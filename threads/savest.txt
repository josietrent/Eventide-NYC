// Staunton Island Condo hideout save pickup thread
{$CLEO .cs}
{$USE CLEO}
{$INCLUDE ../utils/globals.txt}
nop

script_name 'SaveST'

while not Player.IsPlaying($player)
    wait 250
end

// Wait for initial fade-in / save restore to settle
while true
    wait 100
    if and
        not Camera.GetFadingStatus()
        Game.HasSaveGameFinished()
    then
        break
    end
end

// Wait until Staunton Island is unlocked ($BarrierIND == 0)
while $BarrierIND == 1
    wait 250
end

if $SafeBlipST == 0
then
    $SafeBlipST = Blip.AddSpriteForContactPoint(513.55, -627.24, 16.66, RadarSprite.Savehouse)
end

while true
    wait 0

    if or
        Camera.GetFadingStatus()
        not Player.IsPlaying($player)
        $onmission == true
    then
        continue
    end

    if and
        Player.LocateAnyMeans3D($player, 513.55, -627.24, 16.66, 1.0, 2.0, 0.75, True)
        Game.HasSaveGameFinished()
    then
        Game.ActivateSaveMenu()
        Player.SetControl($player, False)
        Player.SetCoordinates($player, 516.50, -627.24, 15.62)
        Player.SetHeading($player, 270.0)

        :WAIT_FOR_SAVE_ST
        wait 0
        if
            Game.HasSaveGameFinished()
        then
            wait 250
            Player.SetControl($player, True)
        else
            jump @WAIT_FOR_SAVE_ST
        end
        continue
    end
end

//-------------Mission 6 (triggered by :Sap in main.txt)---------------
:Sap
script_name 'Sap'

Char Kenji = 0
Char Guard1 = 0
Char Guard2 = 0
Char Guard3 = 0
Char Guard4 = 0
Char Guard5 = 0
Char Guard6 = 0
Pickup Sap1 = 0
Pickup Sap2 = 0
Pickup Sap3 = 0
Blip SapBlip1 = 0
Blip SapBlip2 = 0
Blip SapBlip3 = 0
Sphere ReturnSphere = 0
int SapCount = 0

$onmission = 1
$OnSapMission = 1
$SapMissionDone = 0
wait 100

Streaming.RequestModel(#GANG03)
Streaming.RequestModel(#BAT)
Streaming.RequestModel(#COLT45)
Streaming.RequestModel(#SHOTGUN)
Streaming.LoadSpecialCharacter(1, 'KENJI')
wait 500
Streaming.LoadAllModelsNow()
wait 500

while true
    wait 0
    if and
        Streaming.HasModelLoaded(#GANG03)
        Streaming.HasModelLoaded(#BAT)
        Streaming.HasModelLoaded(#COLT45)
        Streaming.HasModelLoaded(#SHOTGUN)
        Streaming.HasSpecialCharacterLoaded(1)
    then
        break
    end
end

Hud.SwitchWidescreen(True)
wait 1000
Player.SetControl($player, False)

// Create Kenji on the South side of the Head Radio roof
World.ClearArea(1018.75, -68.50, 19.70, 8.0, True)
Kenji = Char.Create(PedType.Special, #SPECIAL01, 1020.50, -68.50, 19.70)
Char.SetObjNoObj(Kenji)
Char.SetHeading(Kenji, 90.0)

// Position Toni on the South side of the Head Radio roof facing Kenji
Player.SetCoordinates($player, 1017.00, -68.50, 19.70)
Player.SetHeading($player, 270.0)
wait 0
Char.SetHeading(Kenji, 90.0)
Char.SetObjWaitOnFoot(Kenji)
Char.TurnToFacePlayer(Kenji, $player)

Camera.SetFixedPosition(1015.00, -70.00, 20.80, 0.0, 0.0, 0.0)
Camera.PointAtChar(Kenji, CameraMode.FollowPed, SwitchType.JumpCut)
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(1500, Fade.In)
wait 1500

// Kenji Dialogue Sequence with skip check
Text.Print('SAP1', 4000, 1)
TIMERA = 0
while TIMERA < 4000
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Briefing_Skip
    end
end

Text.Print('SAP2', 4500, 1)
TIMERA = 0
while TIMERA < 4500
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Briefing_Skip
    end
end

Text.Print('SAP3', 4500, 1)
TIMERA = 0
while TIMERA < 4500
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Briefing_Skip
    end
end

Text.Print('SAP4', 4500, 1)
TIMERA = 0
while TIMERA < 4500
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Briefing_Skip
    end
end

:Sap_Briefing_Skip
Text.ClearPrints()
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(800, Fade.Out)
wait 800

if not Kenji == 0
then
    Char.RemoveElegantly(Kenji)
    Kenji = 0
end
Streaming.UnloadSpecialCharacter(1)

World.ClearArea(978.25, -51.00, 7.35, 8.0, True)
Player.SetCoordinates($player, 976.50, -51.00, 7.35)
Player.SetHeading($player, 270.0)
Camera.RestoreJumpcut()
Hud.SwitchWidescreen(False)
Camera.DoFade(800, Fade.In)
wait 800
Player.SetControl($player, True)

// Stage 1: Spawn 3 guarded Sap caches at verified Portland outskirts sites (IDE/IPL & packages.txt)
// Site 1: Harwood North Coast path (ind_land150 COL ground Z=2.51, clear of hidden package at 761.59, 137.26)
World.ClearArea(755.53, 140.56, 3.55, 10.0, True)
Sap1 = Pickup.Create(#BRIEFCASE, PickupType.Once, 755.53, 140.56, 3.55)
SapBlip1 = Blip.AddForPickup(Sap1)

Guard1 = Char.Create(PedType.Gang3, #GANG03, 756.00, 125.00, 3.50)
Char.SetHeading(Guard1, 180.0)
Char.GiveWeapon(Guard1, WeaponType.Pistol, 100)
Char.SetCurrentWeapon(Guard1, WeaponType.Pistol)
Char.SetObjKillCharAnyMeans(Guard1, $scplayer)

Guard2 = Char.Create(PedType.Gang3, #GANG03, 758.00, 112.00, 3.50)
Char.SetHeading(Guard2, 180.0)
Char.GiveWeapon(Guard2, WeaponType.BaseballBat, 1)
Char.SetCurrentWeapon(Guard2, WeaponType.BaseballBat)
Char.SetObjKillCharAnyMeans(Guard2, $scplayer)

// Site 2: Portland Beach concrete bluff (indncoast1 COL ground Z=8.28, clear of ocean edge at Y>=26)
World.ClearArea(1503.22, 20.09, 9.32, 10.0, True)
Sap2 = Pickup.Create(#BRIEFCASE, PickupType.Once, 1503.22, 20.09, 9.32)
SapBlip2 = Blip.AddForPickup(Sap2)

Guard3 = Char.Create(PedType.Gang3, #GANG03, 1502.61, 16.36, 9.32)
Char.SetHeading(Guard3, 90.0)
Char.GiveWeapon(Guard3, WeaponType.Pistol, 100)
Char.SetCurrentWeapon(Guard3, WeaponType.Pistol)
Char.SetObjKillCharAnyMeans(Guard3, $scplayer)

Guard4 = Char.Create(PedType.Gang3, #GANG03, 1497.00, 18.00, 9.32)
Char.SetHeading(Guard4, 135.0)
Char.GiveWeapon(Guard4, WeaponType.Shotgun, 40)
Char.SetCurrentWeapon(Guard4, WeaponType.Shotgun)
Char.SetObjKillCharAnyMeans(Guard4, $scplayer)

// Site 3: Callahan Point upper bluff (ind_land019 COL ground Z=13.5..13.8, above Predator dock path)
World.ClearArea(815.44, -1051.81, 14.60, 10.0, True)
Sap3 = Pickup.Create(#BRIEFCASE, PickupType.Once, 815.44, -1051.81, 14.60)
SapBlip3 = Blip.AddForPickup(Sap3)

Guard5 = Char.Create(PedType.Gang3, #GANG03, 811.00, -1056.00, 14.80)
Char.SetHeading(Guard5, 45.0)
Char.GiveWeapon(Guard5, WeaponType.Pistol, 100)
Char.SetCurrentWeapon(Guard5, WeaponType.Pistol)
Char.SetObjKillCharAnyMeans(Guard5, $scplayer)

Guard6 = Char.Create(PedType.Gang3, #GANG03, 818.50, -1056.00, 14.50)
Char.SetHeading(Guard6, 0.0)
Char.GiveWeapon(Guard6, WeaponType.Shotgun, 40)
Char.SetCurrentWeapon(Guard6, WeaponType.Shotgun)
Char.SetObjKillCharAnyMeans(Guard6, $scplayer)

Text.Print('SAP5', 5000, 1)
wait 0
$TIMER = 240000 // 4:00 pre-dawn countdown
Hud.DisplayTimer($TIMER)

while SapCount < 3
    wait 250
    if not Player.IsPlaying($player)
    then
        jump @Sap_Fail
    end

    if $TIMER <= 0
    then
        Text.Print('SAP9', 4000, 1)
        jump @Sap_Fail
    end

    if not SapBlip1 == 0
    then
        if Pickup.HasBeenCollected(Sap1)
        then
            SapCount += 1
            Blip.Remove(SapBlip1)
            SapBlip1 = 0
            Sap1 = 0
            if SapCount == 1
            then
                Text.Print('SAP6', 4000, 1)
            end
            if SapCount == 2
            then
                Text.Print('SAP7', 4000, 1)
            end
        end
    end

    if not SapBlip2 == 0
    then
        if Pickup.HasBeenCollected(Sap2)
        then
            SapCount += 1
            Blip.Remove(SapBlip2)
            SapBlip2 = 0
            Sap2 = 0
            if SapCount == 1
            then
                Text.Print('SAP6', 4000, 1)
            end
            if SapCount == 2
            then
                Text.Print('SAP7', 4000, 1)
            end
        end
    end

    if not SapBlip3 == 0
    then
        if Pickup.HasBeenCollected(Sap3)
        then
            SapCount += 1
            Blip.Remove(SapBlip3)
            SapBlip3 = 0
            Sap3 = 0
            if SapCount == 1
            then
                Text.Print('SAP6', 4000, 1)
            end
            if SapCount == 2
            then
                Text.Print('SAP7', 4000, 1)
            end
        end
    end
end

// All 3 collected! Clear timer, trigger atmospheric shift & return to Kenji

play_audio_stream_1channel "audio/EC/miasma.wav", False, 1.0
Weather.ForceNow(WeatherType.Rainy)
Camera.SetFadingColor(30, 0, 82)
Camera.DoFade(300, Fade.Out)
wait 300
Camera.DoFade(600, Fade.In)
wait 600

// Spawn Kenji back at Harwood contact point for wrap-up
Streaming.LoadSpecialCharacter(1, 'KENJI')
while not Streaming.HasSpecialCharacterLoaded(1)
    wait 0
end
World.ClearArea(1004.42, -49.06, 7.44, 6.0, True)
Kenji = Char.Create(PedType.Special, #SPECIAL01, 1004.42, -49.06, 7.44)
Char.SetHeading(Kenji, 270.0)
Char.SetObjWaitOnFoot(Kenji)

// Reuse SapBlip1 as the return blip
SapBlip1 = Blip.AddForCoord(1004.42, -49.06, 7.44)
Blip.ChangeColor(SapBlip1, BlipColor.Yellow)
ReturnSphere = Sphere.Create(1004.42, -49.06, 7.44, 4.0)
Text.Print('SAP8', 5000, 1)

while true
    wait 250
    if or
        not Player.IsPlaying($player)
        not Char.IsStillAlive(Kenji)
    then
        jump @Sap_Fail
    end

    if Player.LocateAnyMeans3D($player, 1004.42, -49.06, 7.44, 4.0, 4.0, 3.0, False)
    then
        Blip.Remove(SapBlip1)
        SapBlip1 = 0
        Sphere.Remove(ReturnSphere)
        ReturnSphere = 0
        break
    end
end

// Stage 3: Ending cutscene with Kenji on the South side of the Head Radio roof
Player.SetControl($player, False)
Hud.SwitchWidescreen(True)
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(800, Fade.Out)
wait 800

World.ClearArea(1018.75, -68.50, 19.70, 8.0, True)
if Player.IsInAnyCar($player)
then
    Player.WarpFromCarToCoord($player, 1017.00, -68.50, 19.70)
else
    Player.SetCoordinates($player, 1017.00, -68.50, 19.70)
end
Player.SetHeading($player, 270.0)
Char.SetCoordinates(Kenji, 1020.50, -68.50, 19.70)
Char.SetObjNoObj(Kenji)
Char.SetHeading(Kenji, 90.0)
wait 0
Char.SetHeading(Kenji, 90.0)
Char.SetObjWaitOnFoot(Kenji)
Char.TurnToFacePlayer(Kenji, $player)

Camera.SetFixedPosition(1015.00, -70.00, 20.80, 0.0, 0.0, 0.0)
Camera.PointAtChar(Kenji, CameraMode.FollowPed, SwitchType.JumpCut)
Camera.DoFade(800, Fade.In)
wait 800

Text.Print('SAPA', 4000, 1)
TIMERA = 0
while TIMERA < 4000
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Ending_Skip
    end
end

Text.Print('SAPB', 4500, 1)

while TIMERA < 4500
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)

    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Ending_Skip
    end
end

:Sap_Ending_Skip
Text.ClearPrints()
Camera.DoFade(1000, Fade.Out)
wait 1000
Weather.ForceNow(WeatherType.Foggy)
if not Kenji == 0
then
    Char.RemoveElegantly(Kenji)
    Kenji = 0
end
Streaming.UnloadSpecialCharacter(1)
World.ClearArea(978.25, -51.00, 7.35, 8.0, True)
Player.SetCoordinates($player, 976.50, -51.00, 7.35)
Player.SetHeading($player, 270.0)
Camera.RestoreJumpcut()
Hud.SwitchWidescreen(False)
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(1000, Fade.In)
wait 1000

// Mission Success!
Text.PrintWithNumberBig('MISPASS', 25000, 5000, TextStyle.Middle)
Audio.PlayMissionPassedTune(1)
Player.AddScore($player, 25000)
Stat.RegisterMissionPassed('MIS6NAM')
Mission.Finish()
$SapMissionDone = 1
$onmission = 0

// Cleanup
if not Kenji == 0
then
    Char.RemoveElegantly(Kenji)
    Kenji = 0
end
if not Guard1 == 0

    Char.RemoveElegantly(Guard1)
    Guard1 = 0
end

then
    Char.RemoveElegantly(Guard2)
    Guard2 = 0

then
    Char.RemoveElegantly(Guard3)
    Guard3 = 0
end

then
    Char.RemoveElegantly(Guard4)
    Guard4 = 0
end
if not Guard5 == 0
then
    Char.RemoveElegantly(Guard5)
    Guard5 = 0
end
if not Guard6 == 0
then
    Char.RemoveElegantly(Guard6)
    Guard6 = 0
end
Streaming.UnloadSpecialCharacter(1)
Streaming.MarkModelAsNoLongerNeeded(#GANG03)
Streaming.MarkModelAsNoLongerNeeded(#BAT)
Streaming.MarkModelAsNoLongerNeeded(#COLT45)
Streaming.MarkModelAsNoLongerNeeded(#SHOTGUN)
Player.SetControl($player, True)

:Sap_Fail

if not SapBlip1 == 0
then
    Blip.Remove(SapBlip1)
    SapBlip1 = 0
end
if not ReturnSphere == 0

    Sphere.Remove(ReturnSphere)
    ReturnSphere = 0
end
if not SapBlip2 == 0

    Blip.Remove(SapBlip2)
    SapBlip2 = 0

    Blip.Remove(SapBlip3)

    Pickup.Remove(Sap1)
    Sap1 = 0

end

then
    Char.Delete(Guard6)
    Guard6 = 0
end
Streaming.UnloadSpecialCharacter(1)
Streaming.MarkModelAsNoLongerNeeded(#GANG03)
Streaming.MarkModelAsNoLongerNeeded(#BAT)
Streaming.MarkModelAsNoLongerNeeded(#COLT45)

Camera.RestoreJumpcut()

wait 1500
Player.SetControl($player, True)
$OnSapMission = 0
terminate_this_custom_script

{$CLEO .cm}

{$USE CLEO, AUDIO}

{$INCLUDE ../utils/globals.txt}

//-------------Mission 7 (triggered by :Sap in main.txt)---------------
:Sap
script_name 'Sap'

Char Kenji = 0
Char Guard1 = 0
Char Guard2 = 0
Char Guard3 = 0
Char Guard4 = 0
Char Guard5 = 0
Char Guard6 = 0
Pickup Sap1 = 0
Pickup Sap2 = 0
Pickup Sap3 = 0
Blip SapBlip1 = 0
Blip SapBlip2 = 0
Blip SapBlip3 = 0
int SapCount = 0
int TimerLimit = 240000 // 4:00 pre-dawn countdown

$onmission = 1
$OnSapMission = 1
$SapMissionDone = 0
wait 100

Streaming.RequestModel(#GANG03)
Streaming.RequestModel(#BAT)
Streaming.RequestModel(#COLT45)
Streaming.RequestModel(#SHOTGUN)
Streaming.LoadSpecialCharacter(1, 'KENJI')
wait 500
Streaming.LoadAllModelsNow()
wait 500

while true
    wait 0
    if and
        Streaming.HasModelLoaded(#GANG03)
        Streaming.HasModelLoaded(#BAT)
        Streaming.HasModelLoaded(#COLT45)
        Streaming.HasModelLoaded(#SHOTGUN)
        Streaming.HasSpecialCharacterLoaded(1)
    then
        break
    end
end

Hud.SwitchWidescreen(True)
wait 1000
Player.SetControl($player, False)

// Create Kenji at Harwood contact point (1004.42, -49.06, 7.44)
World.ClearArea(1004.42, -49.06, 7.44, 6.0, True)
Kenji = Char.Create(PedType.Special, #SPECIAL01, 1004.42, -49.06, 7.44)
Char.SetHeading(Kenji, 90.0)
Char.SetObjWaitOnFoot(Kenji)

// Position Toni facing Kenji
Player.SetCoordinates($player, 1007.50, -49.06, 7.44)
Player.SetHeading($player, 270.0)

Camera.SetFixedPosition(1009.00, -47.00, 8.50, 0.0, 0.0, 0.0)
Camera.PointAtChar(Kenji, CameraMode.FollowPed, SwitchType.JumpCut)
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(1500, Fade.In)
wait 1500

// Kenji Dialogue Sequence with skip check
Text.Print('SAP1', 4000, 1)
TIMERA = 0
while TIMERA < 4000
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Briefing_Skip
    end
end

Text.Print('SAP2', 4500, 1)
TIMERA = 0
while TIMERA < 4500
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Briefing_Skip
    end
end

Text.Print('SAP3', 4500, 1)
TIMERA = 0
while TIMERA < 4500
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Briefing_Skip
    end
end

Text.Print('SAP4', 4500, 1)
TIMERA = 0
while TIMERA < 4500
    wait 0
    if or
        Pad.IsKeyPressed(KeyCode.LeftShift)
        Pad.IsKeyPressed(KeyCode.Space)
        Pad.IsKeyPressed(KeyCode.F)
    then
        jump @Sap_Briefing_Skip
    end
end

:Sap_Briefing_Skip
Text.ClearPrints()
Camera.RestoreJumpcut()
Hud.SwitchWidescreen(False)
Player.SetControl($player, True)

if not Kenji == 0
then
    Char.RemoveElegantly(Kenji)
    Kenji = 0
end
Streaming.UnloadSpecialCharacter(1)

// Stage 1: Spawn 3 guarded Sap caches at verified Portland outskirts sites (IDE/IPL & packages.txt)
// Site 1: Harwood North Coast (761.59, 137.26, 3.55) - near coastal bollardlight path (760.74, 121.66, 2.47)
World.ClearArea(761.59, 137.26, 3.55, 10.0, True)
Sap1 = Pickup.Create(#BRIEFCASE, PickupType.Once, 761.59, 137.26, 3.55)
SapBlip1 = Blip.AddForPickup(Sap1)

Guard1 = Char.Create(PedType.Gang3, #GANG03, 756.00, 125.00, 2.80)
Char.SetHeading(Guard1, 180.0)
Char.GiveWeapon(Guard1, WeaponType.Pistol, 100)
Char.SetCurrentWeapon(Guard1, WeaponType.Pistol)
Char.SetObjKillCharAnyMeans(Guard1, $scplayer)

Guard2 = Char.Create(PedType.Gang3, #GANG03, 764.00, 128.00, 3.00)
Char.SetHeading(Guard2, 180.0)
Char.GiveWeapon(Guard2, WeaponType.BaseballBat, 1)
Char.SetCurrentWeapon(Guard2, WeaponType.BaseballBat)
Char.SetObjKillCharAnyMeans(Guard2, $scplayer)

// Site 2: Portland Beach (1512.27, 24.67, 5.00) - open beach east of ind_land033
World.ClearArea(1512.27, 24.67, 5.00, 10.0, True)
Sap2 = Pickup.Create(#BRIEFCASE, PickupType.Once, 1512.27, 24.67, 5.00)
SapBlip2 = Blip.AddForPickup(Sap2)

Guard3 = Char.Create(PedType.Gang3, #GANG03, 1504.00, 18.00, 4.50)
Char.SetHeading(Guard3, 90.0)
Char.GiveWeapon(Guard3, WeaponType.Pistol, 100)
Char.SetCurrentWeapon(Guard3, WeaponType.Pistol)
Char.SetObjKillCharAnyMeans(Guard3, $scplayer)

Guard4 = Char.Create(PedType.Gang3, #GANG03, 1509.00, 30.00, 4.80)
Char.SetHeading(Guard4, 135.0)
Char.GiveWeapon(Guard4, WeaponType.Shotgun, 40)
Char.SetCurrentWeapon(Guard4, WeaponType.Shotgun)
Char.SetObjKillCharAnyMeans(Guard4, $scplayer)

// Site 3: Callahan Point Wooded Bluff (805.68, -1061.48, 17.20) - clearing on ind_land129
World.ClearArea(805.68, -1061.48, 17.20, 10.0, True)
Sap3 = Pickup.Create(#BRIEFCASE, PickupType.Once, 805.68, -1061.48, 17.20)
SapBlip3 = Blip.AddForPickup(Sap3)

Guard5 = Char.Create(PedType.Gang3, #GANG03, 811.00, -1056.00, 16.50)
Char.SetHeading(Guard5, 45.0)
Char.GiveWeapon(Guard5, WeaponType.Pistol, 100)
Char.SetCurrentWeapon(Guard5, WeaponType.Pistol)
Char.SetObjKillCharAnyMeans(Guard5, $scplayer)

Guard6 = Char.Create(PedType.Gang3, #GANG03, 802.00, -1055.00, 16.50)
Char.SetHeading(Guard6, 0.0)
Char.GiveWeapon(Guard6, WeaponType.Shotgun, 40)
Char.SetCurrentWeapon(Guard6, WeaponType.Shotgun)
Char.SetObjKillCharAnyMeans(Guard6, $scplayer)

Text.PrintHelp('SAP5')
wait 0
Hud.DisplayTimer(TimerLimit)

while SapCount < 3
    wait 250
    if not Player.IsPlaying($player)
    then
        jump @Sap_Fail
    end

    if TimerLimit <= 0
    then
        Text.Print('SAP9', 4000, 1)
        jump @Sap_Fail
    end

    if not SapBlip1 == 0
    then
        if Pickup.HasBeenCollected(Sap1)
        then
            SapCount += 1
            Blip.Remove(SapBlip1)
            SapBlip1 = 0
            Sap1 = 0
            if SapCount == 1
            then
                Text.PrintHelp('SAP6')
            end
            if SapCount == 2
            then
                Text.PrintHelp('SAP7')
            end
        end
    end

    if not SapBlip2 == 0
    then
        if Pickup.HasBeenCollected(Sap2)
        then
            SapCount += 1
            Blip.Remove(SapBlip2)
            SapBlip2 = 0
            Sap2 = 0
            if SapCount == 1
            then
                Text.PrintHelp('SAP6')
            end
            if SapCount == 2
            then
                Text.PrintHelp('SAP7')
            end
        end
    end

    if not SapBlip3 == 0
    then
        if Pickup.HasBeenCollected(Sap3)
        then
            SapCount += 1
            Blip.Remove(SapBlip3)
            SapBlip3 = 0
            Sap3 = 0
            if SapCount == 1
            then
                Text.PrintHelp('SAP6')
            end
            if SapCount == 2
            then
                Text.PrintHelp('SAP7')
            end
        end
    end
end

// All 3 collected! Clear timer, trigger atmospheric shift & return to Kenji
Hud.ClearTimer(TimerLimit)
play_audio_stream_1channel "audio/EC/miasma.wav", False, 1.0
Weather.ForceNow(WeatherType.Rainy)
Camera.SetFadingColor(30, 0, 82)
Camera.DoFade(300, Fade.Out)
wait 300
Camera.DoFade(600, Fade.In)
wait 600

// Spawn Kenji back at Harwood contact point for wrap-up
Streaming.LoadSpecialCharacter(1, 'KENJI')
while not Streaming.HasSpecialCharacterLoaded(1)
    wait 0
end
World.ClearArea(1004.42, -49.06, 7.44, 6.0, True)
Kenji = Char.Create(PedType.Special, #SPECIAL01, 1004.42, -49.06, 7.44)
Char.SetHeading(Kenji, 90.0)
Char.SetObjWaitOnFoot(Kenji)

// Reuse SapBlip1 as the return blip
SapBlip1 = Blip.AddForCoord(1004.42, -49.06, 7.44)
Blip.ChangeColor(SapBlip1, BlipColor.Yellow)
Text.PrintHelp('SAP8')

while true
    wait 250
    if or
        not Player.IsPlaying($player)
        not Char.IsStillAlive(Kenji)
    then
        jump @Sap_Fail
    end

    if Player.LocateAnyMeans3D($player, 1004.42, -49.06, 7.44, 4.0, 4.0, 3.0, True)
    then
        Blip.Remove(SapBlip1)
        SapBlip1 = 0
        break
    end
end

// Stage 3: Ending cutscene with Kenji
Player.SetControl($player, False)
Hud.SwitchWidescreen(True)
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(800, Fade.Out)
wait 800

if Player.IsInAnyCar($player)
then
    Player.WarpFromCarToCoord($player, 1007.50, -49.06, 7.44)
else
    Player.SetCoordinates($player, 1007.50, -49.06, 7.44)
end
Player.SetHeading($player, 270.0)
Char.SetCoordinates(Kenji, 1004.42, -49.06, 7.44)
Char.SetHeading(Kenji, 90.0)

Camera.SetFixedPosition(1009.00, -47.00, 8.50, 0.0, 0.0, 0.0)
Camera.PointAtChar(Kenji, CameraMode.FollowPed, SwitchType.JumpCut)
Camera.DoFade(800, Fade.In)
wait 800

Text.Print('SAPA', 4000, 1)
wait 4000
Text.Print('SAPB', 4500, 1)
wait 4500
Text.Print('SAPC', 4500, 1)
wait 4500

Camera.DoFade(1000, Fade.Out)
wait 1000
Camera.RestoreJumpcut()
Hud.SwitchWidescreen(False)
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(1000, Fade.In)
wait 1000

// Mission Success!
Text.PrintWithNumberBig('MISPASS', 25000, 5000, TextStyle.Middle)
Audio.PlayMissionPassedTune(1)
Player.AddScore($player, 25000)
Stat.RegisterMissionPassed('MIS7NAM')
Mission.Finish()
$SapMissionDone = 1
$onmission = 0

// Cleanup
if not Kenji == 0
then
    Char.RemoveElegantly(Kenji)
    Kenji = 0
end
if not Guard1 == 0
then
    Char.RemoveElegantly(Guard1)
    Guard1 = 0
end
if not Guard2 == 0
then
    Char.RemoveElegantly(Guard2)
    Guard2 = 0
end
if not Guard3 == 0
then
    Char.RemoveElegantly(Guard3)
    Guard3 = 0
end
if not Guard4 == 0
then
    Char.RemoveElegantly(Guard4)
    Guard4 = 0
end
if not Guard5 == 0
then
    Char.RemoveElegantly(Guard5)
    Guard5 = 0
end
if not Guard6 == 0
then
    Char.RemoveElegantly(Guard6)
    Guard6 = 0
end
Streaming.UnloadSpecialCharacter(1)
Streaming.MarkModelAsNoLongerNeeded(#GANG03)
Streaming.MarkModelAsNoLongerNeeded(#BAT)
Streaming.MarkModelAsNoLongerNeeded(#COLT45)
Streaming.MarkModelAsNoLongerNeeded(#SHOTGUN)
Player.SetControl($player, True)
$OnSapMission = 0
terminate_this_custom_script

:Sap_Fail
$onmission = 0
$SapMissionDone = 0
Text.ClearPrints()
Hud.ClearTimer(TimerLimit)
if not SapBlip1 == 0
then
    Blip.Remove(SapBlip1)
    SapBlip1 = 0
end
if not SapBlip2 == 0
then
    Blip.Remove(SapBlip2)
    SapBlip2 = 0
end
if not SapBlip3 == 0
then
    Blip.Remove(SapBlip3)
    SapBlip3 = 0
end
if not Sap1 == 0
then
    Pickup.Remove(Sap1)
    Sap1 = 0
end
if not Sap2 == 0
then
    Pickup.Remove(Sap2)
    Sap2 = 0
end
if not Sap3 == 0
then
    Pickup.Remove(Sap3)
    Sap3 = 0
end
Text.PrintBigString("Mission Failed!", 3000, TextStyle.Middle)
wait 3000
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(1500, Fade.Out)
wait 1500
if not Kenji == 0
then
    Char.Delete(Kenji)
    Kenji = 0
end
if not Guard1 == 0
then
    Char.Delete(Guard1)
    Guard1 = 0
end
if not Guard2 == 0
then
    Char.Delete(Guard2)
    Guard2 = 0
end
if not Guard3 == 0
then
    Char.Delete(Guard3)
    Guard3 = 0
end
if not Guard4 == 0
then
    Char.Delete(Guard4)
    Guard4 = 0
end
if not Guard5 == 0
then
    Char.Delete(Guard5)
    Guard5 = 0
end
if not Guard6 == 0
then
    Char.Delete(Guard6)
    Guard6 = 0
end
Streaming.UnloadSpecialCharacter(1)
Streaming.MarkModelAsNoLongerNeeded(#GANG03)
Streaming.MarkModelAsNoLongerNeeded(#BAT)
Streaming.MarkModelAsNoLongerNeeded(#COLT45)
Streaming.MarkModelAsNoLongerNeeded(#SHOTGUN)
Camera.RestoreJumpcut()
Hud.SwitchWidescreen(False)
Camera.DoFade(1500, Fade.In)
wait 1500
Player.SetControl($player, True)
$OnSapMission = 0
terminate_this_custom_script

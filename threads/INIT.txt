{$CLEO .cm}

{$USE CLEO, Audio, INI}

{$INCLUDE ../utils/globals.txt}

//-------------Merged Intro + Car Mission (triggered by :Driving in main.txt)---------------
:INIT
script_name 'INIT'

int NoodleVisited = 0
int PlayerNotInStallion = 0
int StallionHealth = 0
Blip CarBlip = 0
Blip PunkNoodles = 0
Blip Destination = 0
Sphere NoodleSphere = 0
Sphere SafehouseSphere = 0

$CarDealerBlip = 0
$OnCarMission = 1
$OnGunMission = 0
$onmission = 1
$APOC_LEVEL = 0
$CarMissionDone = 0
$Intro_Passed = 0
$SprayShop_Active = 0
TIMERA = 10000
wait 0

:INIT_Car
wait 500
if NoodleVisited == 0
then
    Player.GetHealth($player, StallionHealth)
    if StallionHealth > 35
    then
        Player.SetHealth($player, 35)
    end
end
if not $Stallion == 0
then
    if Car.IsStillAlive($Stallion)
    then
        if Player.IsInCar($player, $Stallion)
        then
            Car.MarkAsNoLongerNeeded($Stallion)
        else
            Car.Delete($Stallion)
        end
    else
        Car.Delete($Stallion)
    end
    $Stallion = 0
end
if not CarBlip == 0
then
    Blip.Remove(CarBlip)
    CarBlip = 0
end
World.ClearArea(750.85, 84.69, 3.55, 6.0, True)

Streaming.RequestModel(#STALLION)
Streaming.LoadAllModelsNow()
$Stallion = Car.Create(#STALLION, 750.85, 84.69, 3.55)
Car.SetHeading($Stallion, 270.0)
wait 100
CarBlip = Blip.AddForCar($Stallion)
Blip.ChangeColor(CarBlip, BlipColor.Yellow)
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(1500, Fade.In)
wait 1500
Player.SetControl($player, True)
Text.Print('FINDCAR', 5000, 0)

:Car_Out
while true
    wait 0
    if not Player.IsPlaying($player)
    then
        jump @INIT_Fail
    end

    if or
        not Car.IsStillAlive($Stallion)
        Car.IsInWater($Stallion)
    then
        jump @Car_Dead
    end

    if Player.LocateAnyMeans3D($player, 750.85, 84.69, 3.55, 30.0, 30.0, 2.5, False)
    then
        break
    end
end

jump @Car_Enter

:Car_Dead
Text.Print('CARDIE', 4000, 1)
if not CarBlip == 0
then
    Blip.Remove(CarBlip)
    CarBlip = 0
end
if not PunkNoodles == 0
then
    Blip.Remove(PunkNoodles)
    PunkNoodles = 0
end
if not NoodleSphere == 0
then
    Sphere.Remove(NoodleSphere)
    NoodleSphere = 0
end
if not Destination == 0
then
    Blip.Remove(Destination)
    Destination = 0
end
if not SafehouseSphere == 0
then
    Sphere.Remove(SafehouseSphere)
    SafehouseSphere = 0
end
Hud.ClearTimer($TIMER)
PlayerNotInStallion = 0
if not $Stallion == 0
then
    Car.Delete($Stallion)
    $Stallion = 0
end
if $SprayShop_Active == 1
then
    Game.SetFreeResprays(False)
    Blip.Remove($SprayShopIND)
    $SprayShop_Active = 0
end
wait 500
jump @INIT_Car

:Car_Enter
while not Player.IsInCar($player, $Stallion)
    wait 0
    if not Player.IsPlaying($player)
    then
        jump @INIT_Fail
    end

    if or
        not Car.IsStillAlive($Stallion)
        Car.IsInWater($Stallion)
    then
        jump @Car_Dead
    end

    if Player.IsInCar($player, $Stallion)
    then
        stop_audio_stream_1channel
        break
    end
end

if not CarBlip == 0
then
    Blip.Remove(CarBlip)
    CarBlip = 0
end

if NoodleVisited == 1
then
    jump @INIT_Safehouse
end

:INIT_Drive
Player.SetControl($player, False)
Text.Print('STARVE', 3000, 0)
wait 3000
Player.SetControl($player, True)

:INIT_Noodle
PunkNoodles = Blip.AddSpriteForCoord(1041.73, -653.05, 14.82, RadarSprite.Love)
NoodleSphere = Sphere.Create(1041.73, -653.05, 14.82, 3.0)
Text.Print('NOODLE', 5000, 1)
Text.PrintHelp('NOODLEH')

:INIT_Near
while not Player.LocateAnyMeans3D($player, 1041.73, -653.05, 14.82, 15.0, 15.0, 5.0, False)
    wait 0

    if not Player.IsPlaying($player)
    then
        jump @INIT_Fail
    end

    if or
        not Car.IsStillAlive($Stallion)
        Car.IsInWater($Stallion)
    then
        jump @Car_Dead
    end

    if not Player.IsInCar($player, $Stallion)
    then
        if not PunkNoodles == 0
        then
            Blip.Remove(PunkNoodles)
            PunkNoodles = 0
        end
        if not NoodleSphere == 0
        then
            Sphere.Remove(NoodleSphere)
            NoodleSphere = 0
        end
        CarBlip = Blip.AddForCar($Stallion)
        Text.Print('INCAR', 5000, 0)
        PlayerNotInStallion = 1
        jump @INIT_ReturnCar
    end

    if Car.IsUpsidedown($Stallion)
    then
        jump @INIT_Upside
    end
end

Text.Print('MARKER', 3000, 0) // Drive into the marker.
wait 1000

:INIT_Arrived
while not Player.LocateAnyMeans3D($player, 1041.73, -653.05, 14.82, 2.0, 2.0, 3.0, False)
    wait 0

    if not Player.IsPlaying($player)
    then
        jump @INIT_Fail
    end

    if or
        not Car.IsStillAlive($Stallion)
        Car.IsInWater($Stallion)
    then
        jump @Car_Dead
    end

    if Car.IsUpsidedown($Stallion)
    then
        jump @INIT_Upside
    end

    if not Player.IsInCar($player, $Stallion)
    then
        if not PunkNoodles == 0
        then
            Blip.Remove(PunkNoodles)
            PunkNoodles = 0
        end
        if not NoodleSphere == 0
        then
            Sphere.Remove(NoodleSphere)
            NoodleSphere = 0
        end
        CarBlip = Blip.AddForCar($Stallion)
        Text.Print('INCAR', 5000, 0)
        PlayerNotInStallion = 1
        jump @INIT_ReturnCar
    end
end

if not PunkNoodles == 0
then
    Blip.Remove(PunkNoodles)
    PunkNoodles = 0
end
if not NoodleSphere == 0
then
    Sphere.Remove(NoodleSphere)
    NoodleSphere = 0
end
Sound.AddOneOffSound(1041.73, -653.05, 14.82, ScriptSound.SoundPartMissionComplete)
Player.SetHealth($player, 100)
Text.Print('NOODLE2', 3000, 1)
wait 2000
NoodleVisited = 1
jump @INIT_Safehouse

:INIT_ReturnCar
while PlayerNotInStallion == 1
    wait 0
    if not Player.IsPlaying($player)
    then
        jump @INIT_Fail
    end

    if or
        not Car.IsStillAlive($Stallion)
        Car.IsInWater($Stallion)
    then
        jump @Car_Dead
    end

    if Player.IsInCar($player, $Stallion)
    then
        if not CarBlip == 0
        then
            Blip.Remove(CarBlip)
            CarBlip = 0
        end
        PlayerNotInStallion = 0
        PunkNoodles = Blip.AddSpriteForCoord(1041.73, -653.05, 14.82, RadarSprite.Love)
        NoodleSphere = Sphere.Create(1041.73, -653.05, 14.82, 3.0)
        Text.Print('NOODLE', 5000, 1)
        Text.PrintHelp('NOODLEH')
        jump @INIT_Near
    end
end

:INIT_Safehouse
if $SprayShop_Active == 0
then
    if Destination == 0
    then
        Destination = Blip.AddForCoord(1419.15, -194.40, 49.74)
    end
    if SafehouseSphere == 0
    then
        SafehouseSphere = Sphere.Create(1419.15, -194.40, 49.74, 3.0)
    end
    Text.Print('FINHIDE', 5000, 1) // Find a place to stay for the night.
end

while true
    wait 0

    if not Player.IsPlaying($player)
    then
        jump @INIT_Fail
    end

    if or
        not Car.IsStillAlive($Stallion)
        Car.IsInWater($Stallion)
    then
        jump @Car_Dead
    end

    if Car.IsUpsidedown($Stallion)
    then
        jump @INIT_Upside
    end

    if Player.IsInCar($player, $Stallion)
    then
        if PlayerNotInStallion == 1
        then
            PlayerNotInStallion = 0
            if not CarBlip == 0
            then
                Blip.Remove(CarBlip)
                CarBlip = 0
            end
            Hud.ClearTimer($TIMER)
        end

        Car.GetHealth($Stallion, StallionHealth)
        if StallionHealth < 750
        then
            if $SprayShop_Active == 0
            then
                Text.PrintSoon('JM1_4', 5000, 1) // You're trashing the vehicle! Get it repaired!
                Game.SetFreeResprays(True)
                if not Destination == 0
                then
                    Blip.Remove(Destination)
                    Destination = 0
                end
                if not SafehouseSphere == 0
                then
                    Sphere.Remove(SafehouseSphere)
                    SafehouseSphere = 0
                end
                $SprayShopIND = Blip.AddSpriteForCoord(924.0, -361.0, 10.0, RadarSprite.Spray)
                $SprayShop_Active = 1
            end
        end

        if StallionHealth == 1000
        then
            if $SprayShop_Active == 1
            then
                Game.SetFreeResprays(False)
                Blip.Remove($SprayShopIND)
                Destination = Blip.AddForCoord(1419.15, -194.40, 49.74)
                SafehouseSphere = Sphere.Create(1419.15, -194.40, 49.74, 3.0)
                Text.Print('FINHIDE', 5000, 1) // Find a place to stay for the night.
                $SprayShop_Active = 0
            end
        end

        if and
            $SprayShop_Active == 0
            Player.LocateAnyMeans3D($player, 1419.15, -194.40, 49.74, 3.0, 3.0, 3.0, False)
        then
            Player.SetControl($player, False)
            if not Destination == 0
            then
                Blip.Remove(Destination)
                Destination = 0
            end
            if not SafehouseSphere == 0
            then
                Sphere.Remove(SafehouseSphere)
                SafehouseSphere = 0
            end
            Sound.AddOneOffSound(1419.15, -194.40, 49.74, ScriptSound.SoundPartMissionComplete)
            wait 1500
            Camera.SetFadingColor(40, 0, 70)
            Camera.DoFade(1500, Fade.Out)
            wait 1500
            if not $SafeBlip == 0
            then
                Blip.Remove($SafeBlip)
                $SafeBlip = 0
            end
            if not CarBlip == 0
            then
                Blip.Remove(CarBlip)
                CarBlip = 0
            end
            wait 500
            break
        end
    else
        // Handle Exit Timer
        if PlayerNotInStallion == 0
        then
            $TIMER = 15000
            CarBlip = Blip.AddForCar($Stallion)
            Hud.DisplayTimer($TIMER)
            Text.PrintBig('INCAR', 3000, TextStyle.MiddleSmallerHigher) // Get back in the car!
            PlayerNotInStallion = 1
        end

        if $TIMER <= 0
        then
            jump @INIT_Fail
        end
    end
end

:INIT_Cutscene
while Player.IsPlaying($player)
    wait 0

    if Player.IsInAnyCar($player)
    then
        Player.WarpFromCarToCoord($player, 1452.80, -174.88, 71.17)
    else
        Player.SetCoordinates($player, 1452.80, -174.88, 71.17)
    end

    Player.SetControl($player, False)
    Player.SetHeading($player, 180.0)
    Hud.SwitchWidescreen(True)

    Camera.SetFixedPosition(1423.41, -199.77, 75.43, 0.0, 0.0, 0.0)
    Camera.PointAtPoint(1454.08, -173.18, 55.77, SwitchType.JumpCut)
    Camera.SetFadingColor(40, 0, 70)
    Camera.DoFade(1500, Fade.In)

    TIMERA = 0
    while TIMERA < 1500
        wait 0
        if or
            Pad.IsKeyPressed(KeyCode.LeftShift)
            Pad.IsKeyPressed(KeyCode.Space)
            Pad.IsKeyPressed(KeyCode.F)
        then
            jump @INIT_Pass
        end
    end

    Text.Print('SAVE_1', 5000, 1) // This abandoned house will serve as a nice hideout.
    TIMERA = 0
    while TIMERA < 5000
        wait 0
        if or
            Pad.IsKeyPressed(KeyCode.LeftShift)
            Pad.IsKeyPressed(KeyCode.Space)
            Pad.IsKeyPressed(KeyCode.F)
        then
            jump @INIT_Pass
        end
    end

    Text.Print('SAVE_2', 3000, 1) // For now, at least...
    TIMERA = 0
    while TIMERA < 3000
        wait 0
        if or
            Pad.IsKeyPressed(KeyCode.LeftShift)
            Pad.IsKeyPressed(KeyCode.Space)
            Pad.IsKeyPressed(KeyCode.F)
        then
            jump @INIT_Pass
        end
    end

    Camera.SetFixedPosition(1461.19, -177.64, 55.77, 0.0, 0.0, 0.0)
    Camera.PointAtPoint(1453.64, -173.11, 55.77, SwitchType.JumpCut)
    SafehouseSphere = Sphere.Create(1453.64, -173.11, 55.77, 3.0)
    Text.Print('SAVE_3', 6000, 1) // Walking into this sphere behind the bar will save the game.
    TIMERA = 0
    while TIMERA < 6000
        wait 0
        if or
            Pad.IsKeyPressed(KeyCode.LeftShift)
            Pad.IsKeyPressed(KeyCode.Space)
            Pad.IsKeyPressed(KeyCode.F)
        then
            jump @INIT_Pass
        end
    end

    Text.Print('SAVE_4', 6000, 1) // When saving the game, you will lose any wanted level you may have, and time will advance by six hours.
    TIMERA = 0
    while TIMERA < 6000
        wait 0
        if or
            Pad.IsKeyPressed(KeyCode.LeftShift)
            Pad.IsKeyPressed(KeyCode.Space)
            Pad.IsKeyPressed(KeyCode.F)
        then
            jump @INIT_Pass
        end
    end
    jump @INIT_Pass
end
jump @INIT_Fail

:INIT_Upside
while Car.IsUpsidedown($Stallion)
    wait 0
    Text.Print('UPSIDE', 5000, 0)
    wait 5000
    if not Player.IsPlaying($player)
    then
        jump @INIT_Fail
    end
    if Car.IsUpright($Stallion)
    then
        if NoodleVisited == 1
        then
            jump @INIT_Safehouse
        else
            jump @INIT_Near
        end
    else
        jump @INIT_Fail
    end
end
jump @INIT_Fail

:INIT_Fail
IniFile.WriteString("Entered INIT_Fail block", "Trace.ini", "Enter", "Trace")
$onmission = 0
$Intro_Passed = 0
$CarMissionDone = 0
stop_audio_stream_1channel
Text.ClearPrints()
if not PunkNoodles == 0
then
    Blip.Remove(PunkNoodles)
    PunkNoodles = 0
end
if not CarBlip == 0
then
    Blip.Remove(CarBlip)
    CarBlip = 0
end
if not Destination == 0
then
    Blip.Remove(Destination)
    Destination = 0
end
if not NoodleSphere == 0
then
    Sphere.Remove(NoodleSphere)
    NoodleSphere = 0
end
if not SafehouseSphere == 0
then
    Sphere.Remove(SafehouseSphere)
    SafehouseSphere = 0
end
if $CarDealerBlip > 0
then
    Blip.Remove($CarDealerBlip)
    $CarDealerBlip = 0
end
if $SprayShop_Active == 1
then
    Game.SetFreeResprays(False)
    Blip.Remove($SprayShopIND)
    $SprayShop_Active = 0
end
Hud.ClearTimer($TIMER)
wait 0
Text.PrintBig('MISFAIL', 5000, TextStyle.Middle)
if not $Stallion == 0
then
    if Car.IsStillAlive($Stallion)
    then
        if Player.IsInCar($player, $Stallion)
        then
            Car.MarkAsNoLongerNeeded($Stallion)
        else
            Car.Delete($Stallion)
        end
    else
        Car.Delete($Stallion)
    end
    $Stallion = 0
end
Streaming.MarkModelAsNoLongerNeeded(#STALLION)
wait 500
Player.SetControl($player, True)
$OnCarMission = 0
terminate_this_custom_script

:INIT_Pass
Text.ClearPrints()
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(1500, Fade.Out)
wait 1500
if not SafehouseSphere == 0
then
    Sphere.Remove(SafehouseSphere)
    SafehouseSphere = 0
end
$SafeBlip = Blip.AddSpriteForContactPoint(1454.08, -173.18, 55.77, RadarSprite.Savehouse)
Hud.SwitchWidescreen(False)
wait 500
Camera.RestoreJumpcut()
Streaming.MarkModelAsNoLongerNeeded(#STALLION)
if not $Stallion == 0
then
    Car.MarkAsNoLongerNeeded($Stallion)
    $Stallion = 0
end
Text.Print('GTOROOF', 8000, 0)
Camera.DoFade(1500, Fade.In)
wait 1500
Audio.PlayMissionPassedTune(1)
Player.AddScore($player, 1000)
Player.AlterWantedLevel($player, 0)
Text.PrintWithNumberBig('MISPASS', 1000, 5000, TextStyle.Middle)
wait 3000

// REWARD
Player.SetControl($player, True)
Player.SetNeverGetsTired($player, True)
wait 2000
$CarDealerBlip = Blip.AddSpriteForContactPoint(1233.78, -129.02, 15.05, RadarSprite.Tony)
Text.Print('TOMMY', 5000, 1)
wait 4000

Stat.RegisterMissionPassed('MIS0NAM')
Mission.Finish()
$APOC_LEVEL = 1
$Intro_Passed = 1
$CarMissionDone = 1
$OnCarMission = 0
$onmission = 0
terminate_this_custom_script

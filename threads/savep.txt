// Barn hideout save pickup & Mansion roof access thread
:SaveP
script_name 'SaveP'
0@ = 0 // RoofSphere

while true
    wait 0

    if or
        Camera.GetFadingStatus()
        not Player.IsPlaying($player)
        $onmission == true
    then
        if not 0@ == 0
        then
            Sphere.Remove(0@)
            0@ = 0
        end
        continue
    end

    if and
        Player.LocateAnyMeans3D($player, 1453.64, -173.11, 55.77, 1.0, 2.0, 0.75, True)
        Game.HasSaveGameFinished()
    then
        if not 0@ == 0
        then
            Sphere.Remove(0@)
            0@ = 0
        end
        Game.ActivateSaveMenu()
        Player.SetControl($player, False)
        Player.SetCoordinates($player, 1454.55, -187.22, 54.44)
        Player.SetHeading($player, 180)

        :WAIT_FOR_SAVE
        wait 0
        if
            Game.HasSaveGameFinished()
        then
            Player.SetControl($player, True)
        else
            jump @WAIT_FOR_SAVE
        end
        continue
    end

    // Show 1.0-wide Mansion roof warp sphere when within 20 units of the Mansion
    if Player.LocateAnyMeans3D($player, 1478.0, -185.0, 55.0, 20.0, 20.0, 15.0, False)
    then
        if 0@ == 0
        then
            0@ = Sphere.Create(1489.06, -185.49, 54.24, 1.0)
        end

        if Player.LocateOnFoot3D($player, 1489.06, -185.49, 54.24, 1.0, 1.0, 1.5, False)
        then
            Sphere.Remove(0@)
            0@ = 0
            Player.SetControl($player, False)
            Camera.SetFadingColor(40, 0, 70)
            Camera.DoFade(500, Fade.Out)
            wait 500
            Player.SetCoordinates($player, 1452.80, -174.88, 71.17)
            Player.SetHeading($player, 180.0)
            Camera.DoFade(500, Fade.In)
            wait 500
            Player.SetControl($player, True)
        end
    else
        if not 0@ == 0
        then
            Sphere.Remove(0@)
            0@ = 0
        end
    end
end

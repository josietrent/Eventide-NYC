:PACKAGE
script_name 'PACKAGE'

$Packages = Game.GetCollectablesCollected()
$PackIndex = $Packages
$BlipActive = 1
2@ = 0
$PackBlip = 0

while true
    wait 250
    while not Player.IsPlaying($player)
        wait 250
    end

    if $PackHunt == 0
    then
        if $PackBlip > 0
        then
            Blip.Remove($PackBlip)
            $PackBlip = 0
        end
        terminate_this_script
    end

    $Packages = Game.GetCollectablesCollected()

    // ==========================================
    // MANUAL INDEX OVERRIDE SNIFFER (COMMA / PERIOD)
    // Only works and updates blip if tracker is running ($PackHunt == 1)
    // ==========================================
    if and
        $PackHunt == 1
        Pad.IsKeyPressed(KeyCode.OemComma)
    then
        if $PackIndex > 0
        then
            $PackIndex -= 1
            $BlipActive = 1
            $CurrentPackages = $PackIndex
            $CurrentPackages += 1
            Text.PrintHelpFormatted("Currently tracked package changed to ~y~Package %d~w~ of %d.", $CurrentPackages, PACKAGE_COUNT)
        else
            Text.PrintHelpFormatted("Already tracking ~y~Package 1~w~ of %d.", PACKAGE_COUNT)
        end
        wait 300
    end

    if and
        $PackHunt == 1
        Pad.IsKeyPressed(KeyCode.OemPeriod)
    then
        if and
            $BarrierIND == 1
            $PackIndex >= 27
        then
            Text.PrintHelp('PACK_ST')
        else
            if and
                $BarrierCOM == 1
                $PackIndex >= 54
            then
                Text.PrintHelp('PACK_SH')
            else
                $CurrentPackages = PACKAGE_COUNT
                $CurrentPackages -= 1
                if $PackIndex < $CurrentPackages
                then
                    $PackIndex += 1
                    $BlipActive = 1
                    0@ = 1
                    if and
                        $BarrierIND == 1
                        $PackIndex >= 27
                    then
                        0@ = 0
                    end
                    if and
                        $BarrierCOM == 1
                        $PackIndex >= 54
                    then
                        0@ = 0
                    end
                    if 0@ == 1
                    then
                        $CurrentPackages = $PackIndex
                        $CurrentPackages += 1
                        Text.PrintHelpFormatted("Currently tracked package changed to ~y~Package %d~w~ of %d.", $CurrentPackages, PACKAGE_COUNT)
                    end
                else
                    Text.PrintHelpFormatted("Already tracking ~y~Package %d~w~ of %d.", PACKAGE_COUNT, PACKAGE_COUNT)
                end
            end
        end
        wait 300
    end

    // ==========================================
    // BLIP MANAGEMENT
    // Only creates/shows a blip while $PackHunt == 1
    // ==========================================
    if $PackHunt == 1
    then
        if $Packages < PACKAGE_COUNT
        then
            if and
                $PackIndex >= 27
                $BarrierIND == 1
            then
                if not 2@ == 1
                then
                    2@ = 1
                    if $PackBlip > 0
                    then
                        Blip.Remove($PackBlip)
                        $PackBlip = 0
                    end
                    Text.PrintHelp('PACK_ST')
                end
            else
                if and
                    $PackIndex >= 54
                    $BarrierCOM == 1
                then
                    if not 2@ == 2
                    then
                        2@ = 2
                        if $PackBlip > 0
                        then
                            Blip.Remove($PackBlip)
                            $PackBlip = 0
                        end
                        Text.PrintHelp('PACK_SH')
                    end
                else
                    if not 2@ == 0
                    then
                        2@ = 0
                        $BlipActive = 1
                    end

                    GetPackageCoords()

                    if $BlipActive == 1
                    then
                        $BlipActive = 0

                        if $PackBlip > 0
                        then
                            Blip.Remove($PackBlip)
                            $PackBlip = 0
                        end

                        if not $PackCoordX == 0.0
                        then
                            $PackBlip = Blip.AddForCoord($PackCoordX, $PackCoordY, $PackCoordZ)
                            Blip.ChangeColor($PackBlip, BlipColor.Red)
                        end
                    end
                end
            end
        else
            if $PackBlip > 0
            then
                Blip.Remove($PackBlip)
                $PackBlip = 0
            end
            $PackHunt = 0
            terminate_this_script
        end
    else
        if $PackBlip > 0
        then
            Blip.Remove($PackBlip)
            $PackBlip = 0
        end
        terminate_this_script
    end
end

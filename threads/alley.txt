:Alley_CheckCarHealth
Car.GetHealth(Cheetah, CheetahHealth)
if CheetahHealth < 300
then
    jump @Alley_Fail
end

if CheetahHealth < LastValidHealth
then
    LastValidHealth = CheetahHealth
end

if Car.Locate3D(Cheetah, 1435.625, -183.4375, 52.1875, 8.0, 4.5, 3.5, False)
then
    if or
        NeedsSpray == 1
        LastValidHealth < 800
    then
        if FreebieWarned == 0
        then
            Text.PrintBigString("Hey! No Freebies!", 3000, TextStyle.MiddleSmaller)
            FreebieWarned = 1
        end
        if CheetahHealth > LastValidHealth
        then
            Car.SetHealth(Cheetah, LastValidHealth)
            CheetahHealth = LastValidHealth
            Text.PrintBigString("Hey! No Freebies!", 3000, TextStyle.MiddleSmaller)
        end
    end
else
    FreebieWarned = 0
    if and
        Car.Locate3D(Cheetah, 924.0, -361.0, 10.0, 15.0, 15.0, 6.0, False)
        CheetahHealth > LastValidHealth
    then
        LastValidHealth = CheetahHealth
        if CheetahHealth >= 900
        then
            Warn800Printed = 0
        end
    end
end

if and
    CheetahHealth < 500
    NeedsSpray == 0
then
    NeedsSpray = 1
    Warn800Printed = 1
    if not CarBlip == 0
    then
        Blip.Remove(CarBlip)
        CarBlip = 0
    end
    if not Destination == 0
    then
        Blip.Remove(Destination)
        Destination = 0
    end
    SprayBlip = Blip.AddSpriteForCoord(924.0, -361.0, 10.0, RadarSprite.Spray)
    Text.PrintString("Get this vehicle resprayed before you wreck it!", 5000)
    return
end

    CheetahHealth >= 900

    Warn800Printed = 0

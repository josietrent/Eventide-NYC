{$CLEO .cm}

{$USE CLEO}

{$INCLUDE ../utils/globals.txt}

//-------------Mission 3 (triggered by :Stairs)---------------
:Alley
script_name 'Alley'

Char Luigi = 0
Car Cheetah = 0
Blip CarBlip = 0
Blip Destination = 0
int InVehTimer = 0

$onmission = 1
$OnAlleyMission = 1
$AlleyMissionDone = 0
wait 100

:Alley_10
Streaming.LoadSpecialCharacter(1, 'LUIGI')
Streaming.RequestModel(#CHEETAH)
wait 500
Streaming.LoadAllModelsNow()

:Alley_20
while not Streaming.HasSpecialCharacterLoaded(1)
    wait 0
    if and
        Streaming.HasModelLoaded(#CHEETAH)
        Streaming.HasSpecialCharacterLoaded(1)
    then
        break
    else
        continue
    end
end

:Alley_101
Hud.SwitchWidescreen(True)
wait 1000
Player.SetControl($player, False)
Player.SetCoordinates($player, 1322.34, -476.86, 49.75)
Player.SetCurrentWeapon($player, WeaponType.Pistol)
Luigi = Char.Create(PedType.Special, #SPECIAL01, 1329.52, -452.97, 54.48)
Cheetah = Car.Create(#CHEETAH, 1344.11, -448.69, 49.79)
Car.SetHeading(Cheetah, 90.0)
Camera.SetFixedPosition(1330.05, -469.82, 49.77, 0.0, 0.0, 0.0)
Camera.PointAtChar(Luigi, CameraMode.FollowPed, SwitchType.JumpCut)
wait 500
Camera.DoFade(1500, Fade.In)
wait 1500
Char.SetObjGotoCoordOnFoot(Luigi, 1330.38, -470.49)

:Alley_102
while not Char.LocateAnyMeans2D(Luigi, 1330.30, -465.53, 0.25, 0.25, False)
    wait 0
    if or
        not Player.IsPlaying($player)
        not Char.IsStillAlive(Luigi)
        not Car.IsStillAlive(Cheetah)
    then
        jump @Alley_Fail
    end
    if
        Char.LocateAnyMeans2D(Luigi, 1330.30, -465.53, 0.25, 0.25, False)
    then
        Camera.SetFixedPosition(1325.71, -473.58, 49.75, 0.0, 0.0, 0.0)
        Char.SetObjNoObj(Luigi)
        Char.SetObjWaitOnFoot(Luigi)
        Char.SetHeading(Luigi, 145.0)
        Player.SetHeading($player, 325.0)
        Char.TurnToFacePlayer(Luigi, $player)
        Player.TurnToFaceChar($player, Luigi)
        wait 500
        Text.PrintString("I've never seen you around before. You're new to the city, ain't ya?", 5000)
        wait 5000
        Camera.PointAtChar($scplayer, CameraMode.Fixed, SwitchType.JumpCut)
        wait 3500
        Camera.PointAtChar(Luigi, CameraMode.Fixed, SwitchType.JumpCut)
        Text.PrintString("You don't talk much. What, did a cat cut out your tongue?", 4000)
        wait 4000
        Text.PrintString("That's a nice piece you've got there.", 3500)
        wait 3500
        Text.PrintString("I'm curious... Do you know how to use it?", 4000)
        wait 3500
        Camera.PointAtPlayer($player, CameraMode.Fixed, SwitchType.JumpCut)
        Char.PlayAnimation($scplayer, AnimGroup.Man, 45, 4.0)
        wait 1000
        Camera.PointAtChar(Luigi, CameraMode.Fixed, SwitchType.JumpCut)
        Text.PrintString("Woah there, now! Take it easy, kid! I was just bustin' your balls!", 5000)
        Char.SetWaitState(Luigi, WaitState.PlayAnimHandsUp, 2000)
        wait 2000
        Char.SetObjNoObj($scplayer)
        Char.SetObjWaitOnFoot($scplayer)
        Text.PrintString("I can see clearly that you do, but do you know your way around town?", 6000)
        wait 6000
        Text.PrintString("Come with me! I'll show you around.", 4000)
        Char.SetObjGotoCoordOnFoot(Luigi, 1336.43, -466.82)
        Char.SetRunning(Luigi, True)
        wait 2500
        Camera.DoFade(1500, Fade.Out)
        wait 1500
        Player.SetCoordinates($player, 1330.38, -470.49, 49.75)
        Text.ClearPrints()
        Camera.RestoreJumpcut()
        Hud.SwitchWidescreen(False)
        Camera.DoFade(1000, Fade.In)
        wait 500
        Player.SetControl($player, True)
        Char.SetObjGotoCoordOnFoot(Luigi, 1336.43, -466.82)
        Char.SetRunning(Luigi, True)
        while not Char.LocateAnyMeans2D(Luigi, 1336.43, -466.82, 1.5, 1.5, False)
            wait 0
            if or
                not Player.IsPlaying($player)
                not Char.IsStillAlive(Luigi)
                not Car.IsStillAlive(Cheetah)
            then
                jump @Alley_Fail
            end
        end
        Char.SetObjEnterCarAsPassenger(Luigi, Cheetah)
        break
    end
end

:Alley_103
while true
    wait 0
    if or
        not Player.IsPlaying($player)
        not Char.IsStillAlive(Luigi)
        not Car.IsStillAlive(Cheetah)
    then
        jump @Alley_Fail
    end
    if
        Char.IsInCar(Luigi, Cheetah)
    then
        CarBlip = Blip.AddForCar(Cheetah)
        Player.SetControl($player, True)
        Text.PrintString("Get into Louie's car.", 5000)
        break
    end
end

:Alley_104
while not Player.IsInCar($player, Cheetah)
    wait 0
    if or
        not Player.IsPlaying($player)
        not Char.IsStillAlive(Luigi)
        not Car.IsStillAlive(Cheetah)
    then
        jump @Alley_Fail
    end
    if
        Player.IsInCar($player, Cheetah)
    then
        Text.PrintString("Drive me to Little Italy.", 5000)
        Blip.Remove(CarBlip)
        Destination = Blip.AddForCoord(1123.62, -380.07, 19.28)
        Blip.ChangeColor(Destination, BlipColor.Yellow)
        break
    end
end

:Alley_105
InVehTimer = 0
TIMERA = 0
while true
    wait 0
    if or
        not Player.IsPlaying($player)
        not Char.IsStillAlive(Luigi)
        not Car.IsStillAlive(Cheetah)
    then
        jump @Alley_Fail
    end

    if and
        Player.IsInCar($player, Cheetah)
        Char.IsInCar(Luigi, Cheetah)
        Player.LocateAnyMeans3D($player, 1123.62, -380.07, 19.28, 3.0, 3.0, 3.0, True)
    then
        Blip.Remove(Destination)
        Player.SetControl($player, False)
        Sound.AddOneOffSound(1123.62, -380.07, 19.28, ScriptSound.SoundPartMissionComplete)
        Text.PrintString("Take Louie home.", 5000)
        Destination = Blip.AddForCoord(1214.89, -327.47, 26.07)
        Blip.ChangeColor(Destination, BlipColor.Yellow)
        Player.SetControl($player, True)
        break
    end

    if
        not Player.IsInCar($player, Cheetah)
    then
        if or
            InVehTimer == 0
            TIMERA >= 5000
        then
            Text.PrintNow('IN_VEH', 5000, 1)
            TIMERA = 0
            InVehTimer = 1
        end
    else
        InVehTimer = 0
    end
end

:Alley_106
InVehTimer = 0
TIMERA = 0
while true
    wait 0
    if or
        not Player.IsPlaying($player)
        not Char.IsStillAlive(Luigi)
        not Car.IsStillAlive(Cheetah)
    then
        jump @Alley_Fail
    end

    if and
        Player.IsInCar($player, Cheetah)
        Char.IsInCar(Luigi, Cheetah)
        Player.LocateAnyMeans3D($player, 1214.89, -327.47, 26.07, 3.0, 3.0, 3.0, True)
    then
        Blip.Remove(Destination)
        Player.SetControl($player, False)
        Hud.SwitchWidescreen(True)
        Camera.DoFade(1500, Fade.Out)
        wait 1500
        Player.ClearWantedLevel($player)
        Camera.DoFade(1500, Fade.In)
        wait 1500
        Text.PrintString("Not bad. You're good at following directions, and you seem pretty handy.", 5000)
        wait 5000
        Text.PrintString("Meet me back here later. I might have some work for you.", 6000)
        wait 6000
        Camera.DoFade(1500, Fade.Out)
        wait 1500
        if
            Streaming.HasModelLoaded(#UZI)
        then
            Char.GiveWeapon(Luigi, WeaponType.Uzi, 500)
        else
            Streaming.RequestModel(#UZI)
            wait 500
            Streaming.LoadAllModelsNow()
            wait 0
            Char.GiveWeapon(Luigi, WeaponType.Uzi, 500)
        end
        Hud.SwitchWidescreen(False)
        Player.ClearWantedLevel($player)
        Camera.DoFade(1500, Fade.In)
        wait 1500
        Char.SetObjLeaveCar(Luigi, Cheetah)
        Text.PrintBigString("Mission Passed!", 5000, TextStyle.Middle)
        Audio.PlayMissionPassedTune(1)
        Blip.Remove($LuigiBlip)
        Mission.Finish()
        $AlleyMissionDone = 1
        $onmission = 0
        wait 2000
        Text.PrintString("Now beat it, kid!", 5000)
        Player.SetControl($player, True)
        Char.SetObjDestroyCar(Luigi, Cheetah)
        wait 1000
        break
    end

    if
        not Player.IsInCar($player, Cheetah)
    then
        if or
            InVehTimer == 0
            TIMERA >= 5000
        then
            Text.PrintNow('IN_VEH', 5000, 1)
            TIMERA = 0
            InVehTimer = 1
        end
    else
        InVehTimer = 0
    end
end

:Alley_Pass
$LuigiBlip = Blip.AddSpriteForContactPoint(1214.89, -327.47, 26.07, RadarSprite.Luigi)
$KenjiBlip = Blip.AddSpriteForContactPoint(1004.42, -49.06, 7.44, RadarSprite.Kenji)
Text.PrintHelpString("A new Point of Contact is available.")
if
    Char.IsStillAlive(Luigi)
then
    Char.MarkAsNoLongerNeeded(Luigi)
end
if
    Car.IsStillAlive(Cheetah)
then
    Car.MarkAsNoLongerNeeded(Cheetah)
end
Streaming.UnloadSpecialCharacter(1)
Streaming.MarkModelAsNoLongerNeeded(#CHEETAH)
Streaming.MarkModelAsNoLongerNeeded(#UZI)
$OnAlleyMission = 0
terminate_this_custom_script

:Alley_Fail
//Mission.Fail()
$onmission = 0
$AlleyMissionDone = 0
Text.ClearPrints()
Blip.Remove(CarBlip)
Blip.Remove(Destination)
Text.PrintBigString("Mission Failed!", 3000, TextStyle.Middle)
wait 3000
Camera.DoFade(1500, Fade.Out)
wait 1500
if
    Player.IsPlaying($player)
then
    if
        Player.IsInAnyCar($player)
    then
        Player.WarpFromCarToCoord($player, 1336.43, -466.82, 48.76)
        Player.SetHeading($player, 180.0)
    else
        Player.SetCoordinates($player, 1336.43, -466.82, 48.76)
        Player.SetHeading($player, 180.0)
    end
end
if
    Car.IsStillAlive(Cheetah)
then
    Car.Delete(Cheetah)
end
if
    Char.IsStillAlive(Luigi)
then
    Char.Delete(Luigi)
end
Streaming.UnloadSpecialCharacter(1)
Camera.RestoreJumpcut()
Hud.SwitchWidescreen(False)
wait 500
Camera.DoFade(1500, Fade.In)
wait 1500
Player.SetControl($player, True)
$OnAlleyMission = 0
terminate_this_custom_script

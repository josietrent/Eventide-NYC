{$CLEO .cs}
{$USE CLEO, ini}
{$INCLUDE ../utils/globals.txt}
nop

Blip PorterTunnel = 0
int TunnelLaunched = 0

// Startup Gate: Wait until the game and player are fully active before spawning anything
while not Player.IsPlaying($player)
    wait 250
end

wait 1000 // Brief buffer to ensure world streaming is settled

// Relaunch Porter Tunnel guard after save reload if bridge help already ran and Staunton is still locked
if and
    $BridgeHelpPlayed == 1
    $BarrierIND == 1
then
    TunnelLaunched = 1
    stream_custom_script "tunnel.cm"
end

if
    $HelpPickupsSpawned == 0
then
    // Spawn bribes
    $HelpBribe1 = Pickup.Create(#BRIBE, PickupType.OnStreet, 1125.60, -251.54, 17.66)
    $HelpBribe2 = Pickup.Create(#BRIBE, PickupType.OnStreet, 953.71, -423.23, 15.25)
    $HelpBribe3 = Pickup.Create(#BRIBE, PickupType.OnStreet, 820.21, -1076.76, 6.82)
    $HelpBribe4 = Pickup.Create(#BRIBE, PickupType.OnStreet, 1148.39, -923.51, 14.71)
    $HelpBribe5 = Pickup.Create(#BRIBE, PickupType.OnStreet, 985.10, -469.83, 14.93)

    $HelpHealth1 = Pickup.Create(#HEALTH, PickupType.OnStreet, 1399.89, -390.78, 50.56)

    // Info pickups
    $HelpBridgeInfo = Pickup.Create(#INFO, PickupType.Once, 1026.04, -932.74, 14.97) // Callahan Bridge coordinate
    $HelpWastedInfo = Pickup.Create(#INFO, PickupType.Once, 1148.41, -585.33, 15.00) // Hospital coordinate
    $HelpPickupsSpawned = 1
end

:Help
script_name 'Help'

:HelpWait
while true
    wait 250

    if or
        not Player.IsPlaying($player)
        Camera.GetFadingStatus()
    then
        continue
    end

    //====================================================
    // Tutorials allowed both on and off missions
    //====================================================

    // 1. Car Tutorial Check
    if and
        Player.IsInAnyCar($player)
        $CarTutorialPlayed == 0
    then
        if
            Car.IsStillAlive($Canyonero)
        then
            Car.MarkAsNoLongerNeeded($Canyonero)
        end
        PrintHowToDrive()
        continue
    end

    // 1b. On-Foot Portable Radio Help Check
    if and
        $CarTutorialPlayed == 1
        $RadioHelpPlayed == 0
        not Player.IsInAnyCar($player)
    then
        PrintRadioHelp()
        continue
    end

    if
        not Player.CanStartMission($player)
    then
        continue
    end

    //====================================================
    // Mega-switch on freeroam player state for tutorials
    //====================================================

    // 0. Enter/Exit Vehicle Check near $Canyonero at startup
    if and
        $DriveHelpPlayed == 0
        $Intro_Passed == 0
        not Player.IsInAnyCar($player)
    then
        if Car.IsStillAlive($Canyonero)
        then
            if Char.LocateAnyMeansCar3D($scplayer, $Canyonero, 10.0, 10.0, 8.0, False)
            then
                $DriveHelpPlayed = 1
                Text.PrintHelp('DRIVE')
                Car.MarkAsNoLongerNeeded($Canyonero)
                continue
            end
        end
    end

    // 2. Wasted / Death Tutorial Check (Hospital Location Check)
    if and
        Player.LocateAnyMeans3D($player, 1148.41, -585.33, 15.00, 2.0, 2.0, 2.0, False)
        $DeathHelpPlayed == 0
    then
        HowToDie()
        continue
    end

    // 3. Bribe Pickup Help Check
    if $BribeHelpPlayed == 0
    then
        // When the player picks up any of Portland's bribes
        if or
            Pickup.HasBeenCollected($HelpBribe1)
            Pickup.HasBeenCollected($HelpBribe2)
            Pickup.HasBeenCollected($HelpBribe3)
            Pickup.HasBeenCollected($HelpBribe4)
            Pickup.HasBeenCollected($HelpBribe5)
        then
            PrintHowToBribe()
            continue
        end
    end

    // 4. Boat Tutorial Check (avoid III.CLEO.asi's buggy 04A8 IsInAnyBoat dword read at veh+0x5C)
    if
        $BoatHelpPlayed == 0
    then
        if or
            Player.IsInModel($player, #PREDATOR)
            Player.IsInModel($player, #SPEEDER)
            Player.IsInModel($player, #REEFER)
            Player.IsInModel($player, #GHOST)
        then
            // ⛵ I'm on a boat, motherfucker!
            PrintHowToBoat()
            continue
        end
    end

    // 5. Sprint Help Check
    if and
        $CarMissionDone == true
        $SprintHelpPlayed == 0
    then
        PrintHowToSprint()
        continue
    end

    // 6. Callahan Bridge Info Check
    if and
        Player.LocateAnyMeans3D($player, 1026.04, -932.74, 14.97, 2.0, 2.0, 2.0, False)
        $BridgeHelpPlayed == 0
    then
        PrintBridgeHelp()
        continue
    end

    // 7. Completion Check
    if and
        $DriveHelpPlayed == 1
        $BoatHelpPlayed == 1
        $BribeHelpPlayed == 1
        $CarTutorialPlayed == 1
        $DeathHelpPlayed == 1
        $SprintHelpPlayed == 1
        $BridgeHelpPlayed == 1
        $RadioHelpPlayed == 1
    then
        terminate_this_custom_script
    end
end

//=============
// Subroutines
//=============

:PrintBridgeHelp
    $BridgeHelpPlayed = 1
    Text.PrintHelp('CALHAN')
    wait 6000
    Text.PrintHelp('CALHAN2')

    PorterTunnel = Blip.AddForCoord(755.8988, 174.7568, -20.0351)
    Blip.ChangeColor(PorterTunnel, BlipColor.Cyan)

    while not Player.LocateAnyMeans3D($player, 756.7936, 176.8842, -20.4789, 10.0, 10.0, 5.0, False)
        wait 0
    end

    Blip.Remove(PorterTunnel)
    if and
        TunnelLaunched == 0
        $BarrierIND == 1
    then
        TunnelLaunched = 1
        stream_custom_script "tunnel.cm"
    end
return

:PrintHowToDrive
    $DriveHelpPlayed = 1
    $CarTutorialPlayed = 1
    wait 100
    Text.PrintHelp('DRIVE2')
    wait 5000
    if not Player.IsInAnyCar($player)
    then
        Text.ClearHelp()
        return
    end
    Text.PrintHelp('DRIVE3')
    wait 5000
    if not Player.IsInAnyCar($player)
    then
        Text.ClearHelp()
        return
    end
    Text.PrintHelp('DRIVE4')
    wait 5000
    Text.ClearHelp()
return

:PrintRadioHelp
    wait 1000
    if and
        Player.IsPlaying($player)
        not Player.IsInAnyCar($player)
        not Camera.GetFadingStatus()
    then
        $RadioHelpPlayed = 1
        Text.PrintHelp('RADIOH')
    end
return

:PrintHowToBoat
    $BoatHelpPlayed = 1
    wait 100
    Text.PrintHelp('BOAT')
    wait 5000
    Text.PrintHelp('BOAT2')
    wait 5000
    Text.PrintHelp('BOAT3')
    wait 5000
    Text.ClearHelp()
return

:PrintHowToBribe
    $BribeHelpPlayed = 1
    wait 100
    Text.PrintHelp('BRIBE')
    wait 5000
    Text.ClearHelp()
return

:PrintHowToSprint
    $SprintHelpPlayed = 1
    wait 100
    Text.PrintHelp('SPRINT')
    wait 5000
    Text.PrintHelp('SPRINT2')
    wait 5000
return

:HowToDie
    $DeathHelpPlayed = 1
    Player.SetControl($player, False)
    Camera.SetFixedPosition(1116.83, -590.72, 25.27, 0.0, 0.0, 0.0)
    Camera.PointAtPoint(1130.80, -590.75, 23.01, SwitchType.JumpCut)
    wait 1000
    Text.PrintHelpString("When you die, you'll end up back here.")
    wait 4000
    Text.PrintHelpString("You'll lose a bit of cash, but at least you're still alive!")
    wait 4000
    Hud.FlashObject(HudObject.Health)
    Text.PrintHelpString("This is your health. Make sure to keep tabs on it.")
    wait 5000
    Hud.FlashObject(0)
    Camera.SetFixedPosition(1405.12, -390.75, 49.97, 0.0, 0.0, 0.0)
    Camera.PointAtPoint(1400.00, -390.75, 50.0, SwitchType.JumpCut)
    Text.PrintHelpString("If you spend some time looking around, you'll find health pickups. Cherish them.")
    wait 5000
    Camera.RestoreJumpcut()
    Player.SetControl($player, True)
return

{$CLEO .cs}
nop

Blip PorterTunnel = 0
int CarTutorialPlayed = false
int BribeHelpPlayed = false
int BoatHelpPlayed = false
int SprintHelpPlayed = false
int DeathHelp = false

// Startup Gate: Wait until the game and player are fully active before spawning anything
while not Player.IsPlaying($player)
    wait 250
end

wait 1000 // Brief buffer to ensure world streaming is settled

// Spawn your pickups with corrected coordinates
Pickup BRIBE_IND1 = Pickup.Create(#BRIBE, PickupType.OnStreet, 1125.60, -251.54, 17.66)
Pickup BRIBE_IND2 = Pickup.Create(#BRIBE, PickupType.OnStreet, 953.71, -423.23, 15.25)
Pickup BRIBE_IND3 = Pickup.Create(#BRIBE, PickupType.OnStreet, 820.21, -1076.76, 6.82)
Pickup BRIBE_IND4 = Pickup.Create(#BRIBE, PickupType.OnStreet, 1148.39, -923.51, 14.71)

Pickup health1 = Pickup.Create(#HEALTH, PickupType.OnStreet, 1399.89, -390.78, 50.56)

// Swapped Spawns:
Pickup CBridgePickup = Pickup.Create(#INFO, PickupType.Once, 1026.04, -932.74, 14.97) // Callahan Bridge coordinate
Pickup WastedInfo    = Pickup.Create(#INFO, PickupType.Once, 1148.41, -585.33, 15.00) // Hospital coordinate

:Help
script_name 'Help'

CarTutorialPlayed = 0
BribeHelpPlayed = 0
BoatHelpPlayed = 0
SprintHelpPlayed = 0
DeathHelp = 0

:HelpWait
while Player.IsPlaying($player)
    wait 250

    if not Player.IsPlaying($player)
    then
        while not Player.IsPlaying($player)
            wait 250
        end
        continue
    end

    if
        not Player.CanStartMission($player)
    then
        break
    end

    //====================================================
    // Mega-switch on player state for specific tutorials
    //====================================================

    // 1. Car Tutorial Check
    if and
        Player.IsInAnyCar($player)
        CarTutorialPlayed == false
    then
        PrintHowToDrive()
        continue
    end

    // 2. Wasted / Death Tutorial Check (Hospital Location Check)
    if and
        Player.LocateAnyMeans3D($player, 1148.41, -585.33, 15.00, 2.0, 2.0, 2.0, False)
        DeathHelp == false
    then
        HowToDie()
        continue
    end

    // 3. Bribe Pickup Help Check
    if BribeHelpPlayed == false
    then
        if or
            Pickup.HasBeenCollected(BRIBE_IND1)
            Pickup.HasBeenCollected(BRIBE_IND2)
            Pickup.HasBeenCollected(BRIBE_IND3)
            Pickup.HasBeenCollected(BRIBE_IND4)
        then
            // FIXME: Why are we seeing the bribe tutorial?
            PrintHowToBribe()
            continue
        end
    end

    // 4. Boat Tutorial Check
    if and
        Player.IsInAnyBoat($player)
        BoatHelpPlayed == false
    then
        // ⛵ I'm on a boat, motherfucker!
        PrintHowToBoat()
        continue
    end

    // 5. Sprint Help Check
    if and
        $CarMissionDone == true
        SprintHelpPlayed == false
    then
        PrintHowToSprint()
        continue
    end

    // 6. Callahan Bridge Info Check
    if
        Player.LocateAnyMeans3D($player, 1026.04, -932.74, 14.97, 2.0, 2.0, 2.0, False)
    then
        PrintBridgeHelp()
        continue
    end

    // 7. Completion Check
    if and
        BoatHelpPlayed == true
        BribeHelpPlayed == true
        CarTutorialPlayed == true
        DeathHelp == true
        SprintHelpPlayed == true
    then
        terminate_this_script
    end
end

// FIXME: Why the fuck not loop?
jump @HelpWait

//=============
// Subroutines
//=============

:PrintBridgeHelp
    Text.PrintHelp('CALHAN')
    wait 6000
    Text.PrintHelp('CALHAN2')

    PorterTunnel = Blip.AddForCoord(755.8988, 174.7568, -20.0351)
    Blip.ChangeColor(PorterTunnel, BlipColor.Cyan)

    while not Player.LocateAnyMeans3D($player, 756.7936, 176.8842, -20.4789, 10.0, 10.0, 5.0, False)
        wait 0
    end

    Blip.Remove(PorterTunnel)
return

:PrintHowToDrive
    CarTutorialPlayed = true
    wait 100
    Text.PrintHelp('DRIVE2')
    wait 5000
    Text.PrintHelp('DRIVE3')
    wait 5000
    Text.PrintHelp('DRIVE4')
    wait 5000
    Text.ClearHelp()
return

:PrintHowToBoat
    BoatHelpPlayed = true
    wait 100
    Text.PrintHelp('BOAT')
    wait 5000
    Text.PrintHelp('BOAT2')
    wait 5000
    Text.PrintHelp('BOAT3')
    wait 5000
    Text.ClearHelp()
return

:PrintHowToBribe
    BribeHelpPlayed = true
    wait 100
    Text.PrintHelp('BRIBE')
    wait 5000
    Text.ClearHelp()
return

:PrintHowToSprint
    SprintHelpPlayed = true
    wait 100
    Text.PrintHelp('SPRINT')
    wait 5000
    Text.PrintHelp('SPRINT2')
    wait 5000
return

:HowToDie
    DeathHelp = true
    Player.SetControl($player, False)
    Camera.SetFixedPosition(1116.83, -590.72, 25.27, 0.0, 0.0, 0.0)
    Camera.PointAtPoint(1130.80, -590.75, 23.01, SwitchType.JumpCut)
    wait 1000
    Text.PrintHelpString("When you die, you'll end up back here.")
    wait 4000
    Text.PrintHelpString("You'll lose a bit of cash, but at least you're still alive!")
    wait 4000
    Hud.FlashObject(HudObject.Health)
    Text.PrintHelpString("This is your health. Make sure to keep tabs on it.")
    wait 5000
    Hud.FlashObject(0)
    Camera.SetFixedPosition(1405.12, -390.75, 49.97, 0.0, 0.0, 0.0)
    Camera.PointAtPoint(1400.00, -390.75, 50.0, SwitchType.JumpCut)
    Text.PrintHelpString("If you spend some time looking around, you'll find health pickups. Cherish them.")
    wait 5000
    Camera.RestoreJumpcut()
    Player.SetControl($player, True)
return

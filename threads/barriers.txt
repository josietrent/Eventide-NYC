:Barrier
script_name 'Barrier'

var
    0@: int
    1@: int
    2@: int
    3@: int
    4@: float
    5@: float
    6@: float
    7@: float
end

/// Whether the player is prevented from entering Staunton
$BarrierIND = true
/// Whether the player is prevented from entering Shoreside
$BarrierCOM = true
0@ = false // PlayerOutside
$BarrierTimer = 0
1@ = 0     // PlayerHealth
2@ = 0     // CollisionWaitTimer
3@ = false // InRestrictedArea (set by :CheckBridgeZone)
4@ = 0.0   // Player X
5@ = 0.0   // Player Y
6@ = 0.0   // Player Z
7@ = 0.0   // Bridge deck min Z threshold

:BarrierSniffer
while $BarrierIND == true
    wait 0

    if
        not Player.IsPlaying($player)
    then
        continue
    end

    gosub @CheckBridgeZone
    if and
        3@ == true
        not Player.IsOnFoot($player)
        0@ == false
    then
        $BarrierTimer = 15000
        Text.PrintHelp('BARPORT')
        0@ = true
        Hud.DisplayTimer($BarrierTimer)
    end

    // State 2: Countdown & Warning phase
    while 0@ == true
        wait 0

        if
            not Player.IsPlaying($player)
        then
            0@ = false
            Hud.ClearTimer($BarrierTimer)
            break
        end

        // If the player left the restricted bridge deck / water, cancel the countdown
        gosub @CheckBridgeZone
        if
            3@ == false
        then
            0@ = false
            Text.ClearHelp()
            Hud.ClearTimer($BarrierTimer)
            break // Exit countdown loop back to monitoring
        end

        // The timer ran out, kill the player
        if
            $BarrierTimer == 0
        then
            Hud.ClearTimer($BarrierTimer)

            // State 3: Penalty Drain Phase
            1@ = Player.GetHealth($player)
            if
                1@ > 20
            then
                1@ -= 10
            else
                1@ -= 1
            end
            Player.SetHealth($player, 1@)
            wait 50

            // ☠ Death penalty with safety collision check
            if
                1@ < 5
            then
                2@ = 0
                Streaming.LoadCollision(LevelName.Industrial)
                while not Streaming.IsCollisionInMemory(LevelName.Industrial)
                    wait 0
                    2@ += 1
                    if 2@ > 200 // Roughly 5 seconds max wait
                    then
                        break
                    end
                end

                Player.SetControl($player, False)
                Camera.DoFade(500, Fade.Out)
                wait 500

                if
                    Player.IsInAnyCar($player)
                then
                    Player.WarpFromCarToCoord($player, 760.70, 92.53, 4.79)
                else
                    Player.SetCoordinates($player, 760.70, 92.53, 4.79)
                end

                wait 500
                Camera.DoFade(500, Fade.In)
                wait 500
                Text.PrintHelp('BARTELP') // Do not set foot on our lands again!
                Player.SetControl($player, True)
                Player.SetHealth($player, 0)
                0@ = false
            end
        end
    end
end

terminate_this_script

// Sets 3@ = true if player leaves IND_ZON in a boat, is in ROADBR2 (Z >= 30.74), or is on the Callahan Bridge deck in ROADBR1.
// Excludes ground/underpass roads beneath the bridge (Z <= 14.08) and water below the main span (Z = 0.0).
:CheckBridgeZone
3@ = false
if and
    Player.IsInAnyBoat($player)
    not Player.IsInZone($player, 'IND_ZON')
then
    3@ = true
    return
end

if
    Player.IsInZone($player, 'ROADBR2')
then
    3@ = true
    return
end

if
    Player.IsInZone($player, 'ROADBR1')
then
    4@, 5@, 6@ = Player.GetCoordinates($player)
    if
        4@ < 789.0
    then
        // Suspended span (deck Z = 37.77..41.92, col bottom Z = 33.45)
        if
            6@ >= 33.0
        then
            3@ = true
        end
    else
        if
            4@ <= 965.0
        then
            // Portland ramp (slope -0.10: deck Z ~= 115.25 - 0.1*X; underpass Z <= 14.08)
            7@ = 4@
            7@ *= -0.1
            7@ += 112.0
            if
                6@ >= 7@
            then
                3@ = true
            end
        else
            // East approach ramp (X = 965..1065) already down near street level (13.8..18.8)
            3@ = true
        end
    end
end
return

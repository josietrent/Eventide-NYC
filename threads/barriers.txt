:Barrier
script_name 'Barrier'

var
    0@: int
    1@: int
    2@: int
    3@: int
    4@: float
    5@: float
    6@: float
    7@: float
    8@: Object
    9@: Object
end

/// Whether the player is prevented from entering Staunton
$BarrierIND = true
/// Whether the player is prevented from entering Shoreside
$BarrierCOM = true
0@ = false // PlayerOutside
$BarrierTimer = 0
1@ = 0     // PlayerHealth
2@ = 0     // CollisionWaitTimer
3@ = false // InRestrictedArea (set by :CheckBridgeZone)
4@ = 0.0   // Player X
5@ = 0.0   // Player Y
6@ = 0.0   // Player Z
7@ = 0.0   // Bridge deck min Z threshold

// Place Staunton-end roadblocks at the foot of Callahan Bridge using generic HELIX_BARRIER (collision always loaded)
Streaming.RequestModel(#HELIX_BARRIER)
Streaming.LoadAllModelsNow()
8@ = Object.CreateNoOffset(#HELIX_BARRIER, 200.0, -921.5, 26.125)
Object.SetHeading(8@, 90.0)
9@ = Object.CreateNoOffset(#HELIX_BARRIER, 200.0, -941.5, 26.125)
Object.SetHeading(9@, 90.0)

:BarrierSniffer
while $BarrierIND == true
    wait 0

    if or
        not Player.IsPlaying($player)
        Camera.GetFadingStatus()
    then
        wait 500
        continue
    end

    // Sleep while save menu is active so a restored save settles for ~2s before zone checks run
    if
        not Game.HasSaveGameFinished()
    then
        wait 2000
        continue
    end

    gosub @CheckBridgeZone
    if and
        3@ == true
        0@ == false
    then
        $BarrierTimer = 15000
        Text.Print('BARPORT', 5000, 1)
        0@ = true
        Hud.DisplayTimer($BarrierTimer)
    else
        // If the player is off the bridge deck in Newport/Staunton or Shoreside, teleport immediately
        if and
            3@ == false
            0@ == false
        then
            if or
                Player.IsInZone($player, 'COM_EAS')
                Player.IsInZone($player, 'COM_ZON')
                Player.IsInZone($player, 'AIRPORT')
                Player.IsInZone($player, 'PROJECT')
                Player.IsInZone($player, 'SWANKS')
                Player.IsInZone($player, 'SUB_IND')
                Player.IsInZone($player, 'BIG_DAM')
                Player.IsInZone($player, 'SUB_ZO2')
                Player.IsInZone($player, 'SUB_ZO3')
            then
                goto @BarrierTeleport
            end
            if
                Player.IsInZone($player, 'SUB_ZON')
            then
                4@, 5@, 6@ = Player.GetCoordinates($player)
                if
                    4@ < -320.0
                then
                    goto @BarrierTeleport
                end
            end
        end
    end

    // State 2: Countdown & Warning phase
    while 0@ == true
        wait 0

        if
            not Player.IsPlaying($player)
        then
            0@ = false
            Hud.ClearTimer($BarrierTimer)
            break
        end

        // If the player left the restricted bridge deck / water, cancel the countdown
        gosub @CheckBridgeZone
        if
            3@ == false
        then
            0@ = false
            Text.ClearHelp()
            Hud.ClearTimer($BarrierTimer)
            if or
                Player.IsInZone($player, 'COM_EAS')
                Player.IsInZone($player, 'COM_ZON')
                Player.IsInZone($player, 'AIRPORT')
                Player.IsInZone($player, 'PROJECT')
                Player.IsInZone($player, 'SWANKS')
                Player.IsInZone($player, 'SUB_IND')
                Player.IsInZone($player, 'BIG_DAM')
                Player.IsInZone($player, 'SUB_ZO2')
                Player.IsInZone($player, 'SUB_ZO3')
            then
                goto @BarrierTeleport
            end
            if
                Player.IsInZone($player, 'SUB_ZON')
            then
                4@, 5@, 6@ = Player.GetCoordinates($player)
                if
                    4@ < -320.0
                then
                    goto @BarrierTeleport
                end
            end
            break // Exit countdown loop back to monitoring
        end

        // The timer ran out, kill the player
        if
            $BarrierTimer == 0
        then
            Hud.ClearTimer($BarrierTimer)

            // State 3: Penalty Drain Phase
            1@ = Player.GetHealth($player)
            if
                1@ > 20
            then
                1@ -= 10
            else
                1@ -= 1
            end
            Player.SetHealth($player, 1@)
            wait 50
            // ☠ Death penalty with safety collision check
            if
                1@ < 5
            then
                goto @BarrierTeleport
            end
        end
    end
end

Object.Delete(8@)
Object.Delete(9@)
if
    $BarrierCOM == false
then
    Streaming.MarkModelAsNoLongerNeeded(#HELIX_BARRIER)
end
World.SetVisibilityOfClosestObjectOfType(1027.25, -933.75, 15.0, 50.0, #INDHELIX_BARRIER, False)
World.SwapNearestBuildingModel(1027.25, -933.75, 15.0, 50.0, #INDHELIX_BARRIER, #LOD_LAND014)
terminate_this_script

:BarrierTeleport
0@ = false
Text.ClearHelp()
Hud.ClearTimer($BarrierTimer)

2@ = 0
Streaming.LoadCollision(LevelName.Industrial)
while not Streaming.IsCollisionInMemory(LevelName.Industrial)
    wait 0
    2@ += 1
    if 2@ > 200 // Roughly 5 seconds max wait
    then
        break
    end
end

Player.SetControl($player, False)
Camera.SetFadingColor(40, 0, 70)
Camera.DoFade(500, Fade.Out)
wait 500

if
    Player.IsInAnyCar($player)
then
    Player.WarpFromCarToCoord($player, 760.70, 92.53, 4.79)
else
    Player.SetCoordinates($player, 760.70, 92.53, 4.79)
end

wait 500
Camera.DoFade(500, Fade.In)
wait 500
Text.Print('BARTELP', 5000, 1) // Do not set foot on our lands again!
Player.SetControl($player, True)
Player.SetHealth($player, 0)
goto @BarrierSniffer

// Sets 3@ = true if player leaves IND_ZON in a boat, is in ROADBR2 (Z >= 30.74), on the Staunton ramp (X = 197..445), in the ROADBR2..ROADBR1 seam (X = 614..618, Z >= 33), or on the Callahan Bridge deck in ROADBR1 (X <= 1005.0).
// Excludes ground/underpass roads beneath the bridge (Z <= 14.08 on Portland, com_roadkb12 on Staunton), water below the main span (Z = 0.0), CBridgePickup (X = 1026.04), and Portland street approaches (X > 1005.0).
:CheckBridgeZone
3@ = false
// Avoid III.CLEO.asi buggy 04A8 IsInAnyBoat (reads dword at veh+0x5C)
if or
    Player.IsInModel($player, #PREDATOR)
    Player.IsInModel($player, #SPEEDER)
    Player.IsInModel($player, #REEFER)
    Player.IsInModel($player, #GHOST)
then
    if
        not Player.IsInZone($player, 'IND_ZON')
    then
        3@ = true
        return
    end
end

if
    Player.IsInZone($player, 'ROADBR2')
then
    3@ = true
    return
end

// Cover the Staunton ramp descending west from ROADBR2 (X = 197..445) while staying above the underpass road (com_roadkb12)
if or
    Player.IsInArea3D($player, 197.0, -952.0, 24.5, 250.0, -910.0, 207.0, False)
    Player.IsInArea3D($player, 250.0, -952.0, 26.5, 445.0, -910.0, 207.0, False)
then
    3@ = true
    return
end

// Cover the ~2.56u zone gap between ROADBR2 (maxX = 614.878) and ROADBR1 (minX = 617.442) on the deck
if
    Player.IsInArea3D($player, 614.0, -959.0, 33.0, 618.0, -908.0, 207.0, False)
then
    3@ = true
    return
end

if
    Player.IsInZone($player, 'ROADBR1')
then
    4@, 5@, 6@ = Player.GetCoordinates($player)
    if
        4@ < 789.0
    then
        // Suspended span (deck Z = 37.77..41.92, col bottom Z = 33.45)
        if
            6@ >= 33.0
        then
            3@ = true
        end
    else
        if
            4@ <= 965.0
        then
            // Portland ramp (slope -0.10: deck Z ~= 115.25 - 0.1*X; underpass Z <= 14.08)
            // Divide by -10.0 instead of multiplying by -0.1 because GTA III quantizes floats to 1/16 (-0.1 -> -0.0625)
            7@ = 4@
            7@ /= -10.0
            7@ += 112.0
            if
                6@ >= 7@
            then
                3@ = true
            end
        else
            if
                4@ <= 1005.0
            then
                // East end of bridge structure (bridge_endbit01 ends at X = 1005.12);
                // excludes CBridgePickup (X = 1026.04) and Portland road approaches (X = 1025..1065.44)
                3@ = true
            end
        end
    end
end
return

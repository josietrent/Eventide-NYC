{$CLEO .cs}
{$USE CLEO}
{$INCLUDE ../utils/globals.txt}
nop

var
    $PoleKnockTotal: int
    $PoleKnockLastModel: int
    $PoleKnockEnabled: int
end

script_name 'POLES'

int lastTotal = 0
int curTotal = 0
int delta = 0
int streak = 0
int mult = 0
int basePay = 0
int payout = 0
int modelId = 0
float px = 0.0
float py = 0.0
float pz = 0.0

while not Player.IsPlaying($player)
    wait 250
end

// Wait for initial fade-in / save restore to settle before syncing knockdown counter
while true
    wait 100
    if and
        not Camera.GetFadingStatus()
        Game.HasSaveGameFinished()
    then
        break
    end
end

$PoleKnockEnabled = 1
lastTotal = $PoleKnockTotal
streak = 0
TIMERA = 10000

while true
    wait 0

    if or
        not Player.IsPlaying($player)
        Camera.GetFadingStatus()
        not Game.HasSaveGameFinished()
    then
        lastTotal = $PoleKnockTotal
        streak = 0
        continue
    end

    curTotal = $PoleKnockTotal
    if curTotal < lastTotal
    then
        lastTotal = curTotal
        streak = 0
        continue
    end

    if curTotal == lastTotal
    then
        if and
            streak > 0
            TIMERA > 7000
        then
            streak = 0
        end
        continue
    end

    delta = curTotal
    delta -= lastTotal
    lastTotal = curTotal

    // Keep story missions & cold-open cutscenes free of ambient side-payout banners
    if or
        $Intro_Passed == 0
        $onmission == 1
    then
        streak = 0
        continue
    end

    if delta > 8
    then
        delta = 8
    end

    modelId = $PoleKnockLastModel
    if or
        modelId == 1394 // doublestreetlght1
        modelId == 1405 // trafficlight1
        modelId == 1352 // Streetlamp2 (double park light)
    then
        basePay = 50
    else
        basePay = 25
    end

    if TIMERA <= 7000
    then
        streak += delta
    else
        streak = delta
    end
    TIMERA = 0

    mult = streak
    if mult > 5
    then
        mult = 5
    end

    payout = basePay
    payout *= mult
    if delta > 1
    then
        payout *= delta
    end

    Player.AddScore($player, payout)
    Char.GetCoordinates($scplayer, px, py, pz)
    Sound.AddOneOffSound(px, py, pz, ScriptSound.SoundPartMissionComplete)

    if mult >= 2
    then
        Text.PrintWith2NumbersNow('POLEC', mult, payout, 2500, 1)
    else
        if modelId == 1405
        then
            Text.PrintWithNumberNow('POLE2', payout, 2500, 1)
        else
            if modelId == 1394
            then
                Text.PrintWithNumberNow('POLE3', payout, 2500, 1)
            else
                if modelId == 1352
                then
                    Text.PrintWithNumberNow('POLE5', payout, 2500, 1)
                else
                    if modelId == 1353
                    then
                        Text.PrintWithNumberNow('POLE4', payout, 2500, 1)
                    else
                        Text.PrintWithNumberNow('POLE1', payout, 2500, 1)
                    end
                end
            end
        end
    end
end

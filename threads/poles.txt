{$CLEO .cs}
{$USE CLEO}
{$INCLUDE ../utils/globals.txt}
nop

var
    $PoleKnockTotal: int
    $PoleKnockLastModel: int
    $PoleKnockEnabled: int
    $TrafficLawActive: int
    $TrafficLawEvent: int
    $TrafficLawFine: int
    $TrafficLawValue: int
end

script_name 'POLES'

int lastTotal = 0
int curTotal = 0
int delta = 0
int streak = 0
int mult = 0
int basePay = 0
int payout = 0
int modelId = 0
int lawEvent = 0
int lawFine = 0
int lawVal = 0
float px = 0.0
float py = 0.0
float pz = 0.0

while not Player.IsPlaying($player)
    wait 250
end

// Wait for initial fade-in / save restore to settle before syncing knockdown counter
while true
    wait 100
    if and
        not Camera.GetFadingStatus()
        Game.HasSaveGameFinished()
    then
        break
    end
end

$PoleKnockEnabled = 1
$TrafficLawActive = 0
$TrafficLawEvent = 0
$TrafficLawFine = 0
$TrafficLawValue = 0
lastTotal = $PoleKnockTotal
streak = 0
TIMERA = 10000

while true
    wait 0

    if or
        not Player.IsPlaying($player)
        Camera.GetFadingStatus()
        not Game.HasSaveGameFinished()
    then
        $TrafficLawActive = 0
        $TrafficLawEvent = 0
        lastTotal = $PoleKnockTotal
        streak = 0
        continue
    end

    // Testing gate: activate once 'BEACH' prints ($ColdOpenDone == 1) and not on a mission
    // (Restore to $BleedMissionDone == 1 and $Intro_Passed == 1 after testing)
    if and
        $ColdOpenDone == 1
        $onmission == 0
    then
        $TrafficLawActive = 1
    else
        $TrafficLawActive = 0
        $TrafficLawEvent = 0
    end

    // Handle queued traffic-law citations / property-damage fines
    if and
        $TrafficLawActive == 1
        $TrafficLawEvent > 0
    then
        lawEvent = $TrafficLawEvent
        lawFine = $TrafficLawFine
        lawVal = $TrafficLawValue
        $TrafficLawEvent = 0
        $TrafficLawFine = 0
        $TrafficLawValue = 0

        if lawFine > 0
        then
            // Reuse delta as curCash and mult as negative fine delta to stay within 16 CLEO locals
            Player.StoreScore($player, delta)
            if delta > 0
            then
                mult = 0
                if lawFine > delta
                then
                    mult -= delta
                else
                    mult -= lawFine
                end
                Player.AddScore($player, mult)
            end
        end

        Char.GetCoordinates($scplayer, px, py, pz)
        Sound.AddOneOffSound(px, py, pz, ScriptSound.SoundPartMissionComplete)

        if lawEvent == 1
        then
            Text.PrintWithNumberNow('TLAW1', lawFine, 4000, 1)
        else
            if lawEvent == 2
            then
                Text.PrintWithNumberNow('TLAW2', lawFine, 4000, 1)
            else
                if lawEvent == 3
                then
                    Text.PrintWithNumberNow('TLAW3', lawFine, 4000, 1)
                else
                    if lawEvent == 4
                    then
                        Text.PrintWith2NumbersNow('TLAW4', lawVal, lawFine, 4000, 1)
                    else
                        if lawEvent == 5
                        then
                            Text.PrintWith2NumbersNow('TLAW5', lawVal, lawFine, 4000, 1)
                        else
                            if lawEvent == 6
                            then
                                Text.PrintWithNumberNow('TLAW6', lawFine, 4000, 1)
                            else
                                if lawEvent == 7
                                then
                                    Text.PrintWithNumberNow('TLAW7', lawFine, 4000, 1)
                                else
                                    if lawEvent == 8
                                    then
                                        Text.PrintWithNumberNow('TLAW8', lawFine, 4000, 1)
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    curTotal = $PoleKnockTotal
    if curTotal < lastTotal
    then
        lastTotal = curTotal
        streak = 0
        continue
    end

    if curTotal == lastTotal
    then
        if and
            streak > 0
            TIMERA > 7000
        then
            streak = 0
        end
        continue
    end

    delta = curTotal
    delta -= lastTotal
    lastTotal = curTotal

    // Keep story missions & cold-open cutscenes free of ambient side-payout banners
    if or
        $Intro_Passed == 0
        $onmission == 1
    then
        streak = 0
        continue
    end

    if delta > 8
    then
        delta = 8
    end

    modelId = $PoleKnockLastModel
    if or
        modelId == 1322 // fire_hydrant
        modelId == 1394 // doublestreetlght1
        modelId == 1405 // trafficlight1
        modelId == 1352 // Streetlamp2 (double park light)
    then
        basePay = 50
    else
        basePay = 25
    end

    if TIMERA <= 7000
    then
        streak += delta
    else
        streak = delta
    end
    TIMERA = 0

    mult = streak
    if mult > 5
    then
        mult = 5
    end

    payout = basePay
    payout *= mult
    if delta > 1
    then
        payout *= delta
    end

    Player.AddScore($player, payout)
    Char.GetCoordinates($scplayer, px, py, pz)
    Sound.AddOneOffSound(px, py, pz, ScriptSound.SoundPartMissionComplete)

    if mult >= 2
    then
        Text.PrintWith2NumbersNow('POLEC', mult, payout, 2500, 1)
    else
        if modelId == 1322
        then
            Text.PrintWithNumberNow('POLE6', payout, 2500, 1)
        else
            if modelId == 1405
            then
                Text.PrintWithNumberNow('POLE2', payout, 2500, 1)
            else
                if modelId == 1394
                then
                    Text.PrintWithNumberNow('POLE3', payout, 2500, 1)
                else
                    if modelId == 1352
                    then
                        Text.PrintWithNumberNow('POLE5', payout, 2500, 1)
                    else
                        if modelId == 1353
                        then
                            Text.PrintWithNumberNow('POLE4', payout, 2500, 1)
                        else
                            Text.PrintWithNumberNow('POLE1', payout, 2500, 1)
                        end
                    end
                end
            end
        end
    end
end

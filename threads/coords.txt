{$CLEO .cs}
{$USE CLEO}
{$INCLUDE ../utils/globals.txt}
nop

script_name 'COORDS'

float x = 0.0
float y = 0.0
float z = 0.0

while not Player.IsPlaying($player)
    wait 250
end

if
    $Intro_Passed == 0
then
    while $ColdOpenDone == 0
        wait 100
        if $Intro_Passed == 1
        then
            break
        end
    end

    if
        $Intro_Passed == 0
    then
        wait 6500
        while true
            if $Intro_Passed == 1
            then
                break
            end
            if and
                Player.IsPlaying($player)
                Player.CanStartMission($player)
                not Camera.GetFadingStatus()
            then
                break
            end
            wait 250
        end

        if
            $Intro_Passed == 0
        then
            Text.PrintHelp('CORHELP')
        end
    end
end

while true
    wait 0
    if not Player.IsPlaying($player)
    then
        continue
    end
    if Pad.IsKeyPressed(KeyCode.Oem5)
    then
        x, y, z = Player.GetCoordinates($player)
        Text.PrintHelpFormatted("Current Coordinates: %.2f, %.2f, %.2f", x, y, z)
        while Pad.IsKeyPressed(KeyCode.Oem5)
            wait 0
        end
    end
end

{$CLEO .cs}
{$USE CLEO}
{$INCLUDE ../utils/globals.txt}
nop

script_name 'WORLD'

int HealthTmp = 0
int NoodleCool = 0

while not Player.IsPlaying($player)
    wait 0
end

Camera.SetFadingColor(40, 0, 70)
if $BarrierIND == 0
then
    World.SetCarDensityMultiplier(0.5)
    World.SetPedDensityMultiplier(0.25)
else
    World.SetCarDensityMultiplier(0.25)
    World.SetPedDensityMultiplier(0.1)
end

// Re-apply after load fade-in settles so engine reset on save reload never leaves density at 1.0
while true
    wait 0
    if and
        not Camera.GetFadingStatus()
        Game.HasSaveGameFinished()
    then
        break
    end
end

Camera.SetFadingColor(40, 0, 70)
if $BarrierIND == 0
then
    World.SetCarDensityMultiplier(0.5)
    World.SetPedDensityMultiplier(0.25)
else
    World.SetCarDensityMultiplier(0.25)
    World.SetPedDensityMultiplier(0.1)
end

// Ensure late-added pickups also spawn when loading an existing savegame
if $AK47Spawn == 0
then
    $AK47Spawn = Pickup.CreateWithAmmo(#AK47, PickupType.OnStreet, 150, 85.13, -986.52, 26.17)
end
if $AK47RoofSpawn == 0
then
    $AK47RoofSpawn = Pickup.CreateWithAmmo(#AK47, PickupType.OnStreet, 150, 1019.40, -26.80, 20.35)
end
if $HEAL_COM6 == 0
then
    $HEAL_COM6 = Pickup.Create(#HEALTH, PickupType.OnStreet, 242.83, -1117.11, 21.11)
end
if $ShotgunSiloSpawn == 0
then
    $ShotgunSiloSpawn = Pickup.CreateWithAmmo(#SHOTGUN, PickupType.OnStreet, 48, 1514.21, -856.59, 27.41)
end
if $LinerunDocksCreated < 2
then
    if $LinerunDocksCreated == 1
    then
        CarGenerator.Switch($LinerunDocksGen, 0)
    end
    // Imported Vice City Packer (model ID 151 in data/default.ide)
    $LinerunDocksGen = CarGenerator.Create(1411.48, -803.43, 11.83, 270.0, 151, -1, -1, False, 0, 0, 0, 10000)
    CarGenerator.Switch($LinerunDocksGen, 101)
    $LinerunDocksCreated = 2
end

// Ensure Newport Condo Stallion exists on older savegames too
if $StallionNewportCreated == 0
then
    $StallionNewportGen = CarGenerator.Create(519.21, -660.87, 16.11, 180.0, #STALLION, -1, -1, False, 0, 0, 0, 10000)
    CarGenerator.Switch($StallionNewportGen, 101)
    $StallionNewportCreated = 1
end

// Ensure Newport Condo Yakuza Stinger exists on older savegames too
if $YakuzaNewportCreated == 0
then
    $YakuzaNewportGen = CarGenerator.Create(517.32, -644.44, 16.14, 180.0, #YAKUZA, -1, -1, False, 0, 0, 0, 10000)
    CarGenerator.Switch($YakuzaNewportGen, 101)
    $YakuzaNewportCreated = 1
end

// Ensure Newport Condo Infernus exists on older savegames too
if $InfernusNewportCreated == 0
then
    $InfernusNewportGen = CarGenerator.Create(487.88, -660.03, 16.12, 180.0, #INFERNUS, -1, -1, False, 0, 0, 0, 10000)
    CarGenerator.Switch($InfernusNewportGen, 101)
    $InfernusNewportCreated = 1
end

// Ensure Staunton Newport Pay 'n' Spray exists on older savegames too
if $SprayGarageCOMCreated == 0
then
    $SprayGarageCOM = Garage.Create(376.25, -496.5, 24.8, 385.6, -490.8, 29.0, GarageType.Respray)
    Garage.SetRotatingDoor($SprayGarageCOM)
    $SprayGarageCOMCreated = 1
    if $BarrierIND == 0
    then
        Garage.Activate($SprayGarageCOM)
        $SprayCOMActive = 1
    else
        Garage.Deactivate($SprayGarageCOM)
        $SprayCOMActive = 0
    end
end

// Ensure Portland Harwood 8-Ball Bomb Shop exists on older savegames too
if $BombGarageINDCreated == 0
then
    $BombGarageIND = Garage.Create(1278.5, -110.75, 13.5, 1285.25, -99.5, 17.5, GarageType.BombShop1)
    $BombGarageINDCreated = 1
    if $SmuggleMissionDone == 1
    then
        Garage.Activate($BombGarageIND)
        $BombINDActive = 1
    else
        Garage.Deactivate($BombGarageIND)
        $BombINDActive = 0
    end
end

// Ensure Aspatria stadium parking lot cars exist on older savegames too (< 2 upgrades saves from Z=15.13->15.125)
if $LotCarsCreated < 2
then
    {$INCLUDE ../utils/lot_cars.txt}
end

// Keep density multipliers, Staunton Pay 'n' Spray state, Harwood Bomb Shop state, short-range Portland/Staunton blips, Act 2 contact markers, and Noodle Shop healing active
while true
    wait 1000
    if Player.IsPlaying($player)
    then
        if $BarrierIND == 0
        then
            World.SetCarDensityMultiplier(0.5)
            World.SetPedDensityMultiplier(0.25)
            if $SprayCOMActive == 0
            then
                Garage.Activate($SprayGarageCOM)
                $SprayCOMActive = 1
            end
            if $StauntonContactsCreated == 0
            then
                $AsukaBlip = Blip.AddSpriteForContactPoint(245.59, -1207.57, 26.67, RadarSprite.Asuka)
                $AsukaSphere = Sphere.Create(245.59, -1207.57, 26.67, 1.5)
                $RayBlip = Blip.AddSpriteForContactPoint(151.72, -757.56, 29.18, RadarSprite.Ray)
                $RaySphere = Sphere.Create(151.72, -757.56, 29.18, 1.5)
                $StauntonContactsCreated = 2
            else
                if $StauntonContactsCreated == 1
                then
                    if not $AsukaBlip == 0
                    then
                        Blip.Remove($AsukaBlip)
                        $AsukaBlip = 0
                    end
                    if not $AsukaSphere == 0
                    then
                        Sphere.Remove($AsukaSphere)
                        $AsukaSphere = 0
                    end
                    $AsukaBlip = Blip.AddSpriteForContactPoint(245.59, -1207.57, 26.67, RadarSprite.Asuka)
                    $AsukaSphere = Sphere.Create(245.59, -1207.57, 26.67, 1.5)
                    $StauntonContactsCreated = 2
                end
            end
        else
            World.SetCarDensityMultiplier(0.25)
            World.SetPedDensityMultiplier(0.1)
            if $SprayCOMActive == 1
            then
                Garage.Deactivate($SprayGarageCOM)
                $SprayCOMActive = 0
            end
        end

        if $SmuggleMissionDone == 1
        then
            if $BombINDActive == 0
            then
                Garage.Activate($BombGarageIND)
                $BombINDActive = 1
            end
        else
            if $BombINDActive == 1
            then
                Garage.Deactivate($BombGarageIND)
                $BombINDActive = 0
            end
        end

        if and
            $BombINDActive == 1
            Streaming.IsCollisionInMemory(LevelName.Industrial)
            not Streaming.IsCollisionInMemory(LevelName.Commercial)
        then
            if $BombBlipIND == 0
            then
                $BombBlipIND = Blip.AddSpriteForCoord(1282.0, -104.5, 14.5, RadarSprite.Bomb)
            end
        else
            if not $BombBlipIND == 0
            then
                Blip.Remove($BombBlipIND)
                $BombBlipIND = 0
            end
        end

        // Portland street weapon pickup short-range blips (filtered by ShortRangeBlips.asi; excludes package rewards)
        if and
            Streaming.IsCollisionInMemory(LevelName.Industrial)
            not Streaming.IsCollisionInMemory(LevelName.Commercial)
        then
            if $PistolBlipIND == 0
            then
                $PistolBlipIND = Blip.AddSpriteForCoord(1400.68, -857.80, 12.01, RadarSprite.Weapon)
            end
            if $ShotgunBlipIND == 0
            then
                $ShotgunBlipIND = Blip.AddSpriteForCoord(1093.14, -239.40, 9.16, RadarSprite.Weapon)
            end
            if $ShotgunSiloBlipIND == 0
            then
                $ShotgunSiloBlipIND = Blip.AddSpriteForCoord(1514.21, -856.59, 27.41, RadarSprite.Weapon)
            end
            if $UziBlipIND == 0
            then
                $UziBlipIND = Blip.AddSpriteForCoord(1143.10, -106.37, 7.47, RadarSprite.Weapon)
            end
            if $AK47RoofBlipIND == 0
            then
                $AK47RoofBlipIND = Blip.AddSpriteForCoord(1019.40, -26.80, 20.35, RadarSprite.Weapon)
            end
        else
            if not $PistolBlipIND == 0
            then
                Blip.Remove($PistolBlipIND)
                $PistolBlipIND = 0
            end
            if not $ShotgunBlipIND == 0
            then
                Blip.Remove($ShotgunBlipIND)
                $ShotgunBlipIND = 0
            end
            if not $ShotgunSiloBlipIND == 0
            then
                Blip.Remove($ShotgunSiloBlipIND)
                $ShotgunSiloBlipIND = 0
            end
            if not $UziBlipIND == 0
            then
                Blip.Remove($UziBlipIND)
                $UziBlipIND = 0
            end
            if not $AK47RoofBlipIND == 0
            then
                Blip.Remove($AK47RoofBlipIND)
                $AK47RoofBlipIND = 0
            end
        end

        // Staunton street weapon pickup short-range blip (filtered by ShortRangeBlips.asi; excludes package rewards)
        if and
            Streaming.IsCollisionInMemory(LevelName.Commercial)
            not Streaming.IsCollisionInMemory(LevelName.Industrial)
        then
            if $AK47BlipCOM == 0
            then
                $AK47BlipCOM = Blip.AddSpriteForCoord(85.13, -986.52, 26.17, RadarSprite.Weapon)
            end
        else
            if not $AK47BlipCOM == 0
            then
                Blip.Remove($AK47BlipCOM)
                $AK47BlipCOM = 0
            end
        end

        // Vice-City-style food vendor healing at Dong's Noodle Shop (Chinatown) in freeroam
        if and
            $Intro_Passed == 1
            $onmission == 0
            Streaming.IsCollisionInMemory(LevelName.Industrial)
        then
            if Player.LocateOnFoot3D($player, 1041.73, -653.05, 14.82, 3.0, 3.0, 2.5, False)
            then
                if NoodleCool == 0
                then
                    Player.GetHealth($player, HealthTmp)
                    if HealthTmp < 100
                    then
                        if Player.IsScoreGreater($player, 9)
                        then
                            Player.AddScore($player, -10)
                        end
                        Player.SetHealth($player, 100)
                        Sound.AddOneOffSound(1041.73, -653.05, 14.82, ScriptSound.SoundPartMissionComplete)
                        Text.Print('NOODLEV', 3500, 1)
                        NoodleCool = 1
                    end
                end
            else
                NoodleCool = 0
            end
        end
    end
end

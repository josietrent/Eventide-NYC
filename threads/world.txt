{$CLEO .cs}
{$USE CLEO}
{$INCLUDE ../utils/globals.txt}
nop

script_name 'WORLD'

while not Player.IsPlaying($player)
    wait 0
end

Camera.SetFadingColor(40, 0, 70)
if $BarrierIND == 0
then
    World.SetCarDensityMultiplier(0.5)
    World.SetPedDensityMultiplier(0.25)
else
    World.SetCarDensityMultiplier(0.25)
    World.SetPedDensityMultiplier(0.1)
end

// Re-apply after load fade-in settles so engine reset on save reload never leaves density at 1.0
while true
    wait 0
    if and
        not Camera.GetFadingStatus()
        Game.HasSaveGameFinished()
    then
        break
    end
end

Camera.SetFadingColor(40, 0, 70)
if $BarrierIND == 0
then
    World.SetCarDensityMultiplier(0.5)
    World.SetPedDensityMultiplier(0.25)
else
    World.SetCarDensityMultiplier(0.25)
    World.SetPedDensityMultiplier(0.1)
end

// Ensure late-added pickups also spawn when loading an existing savegame
if $AK47Spawn == 0
then
    $AK47Spawn = Pickup.CreateWithAmmo(#AK47, PickupType.OnStreet, 150, 85.13, -986.52, 26.17)
end

// Ensure Staunton Newport Pay 'n' Spray exists on older savegames too
if $SprayGarageCOMCreated == 0
then
    $SprayGarageCOM = Garage.Create(376.25, -496.5, 24.8, 385.6, -490.8, 29.0, GarageType.Respray)
    Garage.SetRotatingDoor($SprayGarageCOM)
    $SprayGarageCOMCreated = 1
    if $BarrierIND == 0
    then
        Garage.Activate($SprayGarageCOM)
        $SprayCOMActive = 1
    else
        Garage.Deactivate($SprayGarageCOM)
        $SprayCOMActive = 0
    end
end

// Ensure Aspatria stadium parking lot cars exist on older savegames too (< 2 upgrades saves from Z=15.13->15.125)
if $LotCarsCreated < 2
then
    {$INCLUDE ../utils/lot_cars.txt}
end

// Keep density multipliers, Staunton Pay 'n' Spray state, and Act 2 contact markers enforced across missions, wasted/busted, and engine resets
while true
    wait 1000
    if Player.IsPlaying($player)
    then
        if $BarrierIND == 0
        then
            World.SetCarDensityMultiplier(0.5)
            World.SetPedDensityMultiplier(0.25)
            if $SprayCOMActive == 0
            then
                Garage.Activate($SprayGarageCOM)
                $SprayCOMActive = 1
            end
            if $StauntonContactsCreated == 0
            then
                $AsukaBlip = Blip.AddSpriteForContactPoint(287.92, -658.24, 26.17, RadarSprite.Asuka)
                $AsukaSphere = Sphere.Create(287.92, -658.24, 26.17, 1.5)
                $RayBlip = Blip.AddSpriteForContactPoint(151.72, -757.56, 29.18, RadarSprite.Ray)
                $RaySphere = Sphere.Create(151.72, -757.56, 29.18, 1.5)
                $StauntonContactsCreated = 1
            end
        else
            World.SetCarDensityMultiplier(0.25)
            World.SetPedDensityMultiplier(0.1)
            if $SprayCOMActive == 1
            then
                Garage.Deactivate($SprayGarageCOM)
                $SprayCOMActive = 0
            end
        end
    end
end
